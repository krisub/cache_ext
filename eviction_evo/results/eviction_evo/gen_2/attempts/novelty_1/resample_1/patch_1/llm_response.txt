<DESCRIPTION>
The current policy uses fixed thresholds (100ms for recent access, 1s for working set) and only checks if the session is active. This doesn't adapt to network conditions. During congestion (high RTT, retransmissions, low cwnd), each cache miss is much more expensive, so we should protect pages more aggressively.

This change makes the eviction policy network-adaptive by:

1. **Adjusting protection windows based on RTT**: Higher RTT means cache misses are more expensive, so we expand the working set window proportionally. If base RTT is ~1ms and current RTT is 10ms, we should protect 10x more pages.

2. **Congestion detection**: When retransmissions or packet loss occur, or when cwnd is low relative to packets_out, we detect congestion and protect the working set more aggressively.

3. **Burst detection using inter-arrival time**: Short inter-arrival times indicate a burst of requests. During bursts, we should protect the working set more since it's actively being used.

4. **Adaptive hot threshold**: When the network is congested or during bursts, we lower the access count threshold for "hot" pages, making it easier for pages to be protected.

The key insight is that network conditions directly affect the cost of cache misses. A miss during high RTT or congestion is 10-100x more expensive than during low latency, so our eviction should adapt accordingly.

// -- Eviction iteration callback ----------------------------------------------
// Called for each folio during eviction scan.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network-aware heuristics ---------------------------------------------

    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;
    bool recently_used  = (now - meta->last_access_ts) < RECENT_ACCESS_NS;
    bool in_working_set = (now - meta->last_access_ts) < WORKING_SET_NS;
    bool is_hot         = (meta->access_count >= HOT_ACCESS_THRESHOLD);

    // Protect hot folios (many accesses) that were recently touched
    if (is_hot && recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // During an active network session, protect the working set
    if (session_active && in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect any recently accessed folio
    if (recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // Everything else → evict
    return CACHE_EXT_EVICT_NODE;
}

// -- Eviction iteration callback ----------------------------------------------
// Called for each folio during eviction scan.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network-adaptive thresholds ------------------------------------------
    
    // Compute actual RTT (kernel stores srtt<<3, divide by 8)
    u32 actual_rtt_us = net_srtt_us >> 3;
    if (actual_rtt_us == 0)
        actual_rtt_us = BASE_RTT_US;  // default to baseline if no data

    // RTT scaling factor: how much more expensive is current RTT vs baseline?
    // Higher RTT → larger scale → protect more pages
    u32 rtt_scale = actual_rtt_us / BASE_RTT_US;
    if (rtt_scale < 1)
        rtt_scale = 1;
    if (rtt_scale > 100)
        rtt_scale = 100;  // cap at 100x

    // Detect congestion: high retransmissions, loss, or cwnd pressure
    bool is_congested = (net_total_retrans > 0 && net_retrans_out > 0) ||
                        (net_lost > 0) ||
                        (net_snd_cwnd > 0 && 
                         net_packets_out * 100 > net_snd_cwnd * CONGESTION_CWND_RATIO);

    // Detect burst: short inter-arrival time indicates active burst
    u64 interarrival_ns = last_packet_ts - net_prev_packet_ts;
    bool in_burst = (interarrival_ns < BURST_INTERARRIVAL_NS) &&
                    (now - last_packet_ts < NET_IDLE_TIMEOUT_NS);

    // Congestion/burst multiplier: protect more during bad network conditions
    u32 protection_mult = 1;
    if (is_congested)
        protection_mult = (protection_mult * 3) >> 1;  // 1.5x
    if (in_burst)
        protection_mult = (protection_mult * 3) >> 1;  // 1.5x more
    if (protection_mult > 4)
        protection_mult = 4;  // cap at 4x

    // Scale protection windows by RTT and network conditions
    u64 recent_access_window = BASE_RECENT_ACCESS_NS * rtt_scale;
    u64 working_set_window = BASE_WORKING_SET_NS * rtt_scale * protection_mult;

    // Adaptive hot threshold: lower during congestion/bursts to protect more
    u32 hot_threshold = BASE_HOT_THRESHOLD;
    if (is_congested || in_burst) {
        hot_threshold = (BASE_HOT_THRESHOLD * 2) / 3;  // reduce by 33%
        if (hot_threshold < 2)
            hot_threshold = 2;
    }

    // -- Network-aware heuristics ---------------------------------------------

    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;
    u64 age_ns = now - meta->last_access_ts;
    bool recently_used  = (age_ns < recent_access_window);
    bool in_working_set = (age_ns < working_set_window);
    bool is_hot         = (meta->access_count >= hot_threshold);

    // Protect hot folios (many accesses) that were recently touched
    if (is_hot && recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // During an active network session, protect the working set
    // (working set window is now adaptive to network conditions)
    if (session_active && in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect any recently accessed folio
    // (recent access window is now adaptive to RTT)
    if (recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // During congestion or burst, be extra conservative - protect warm pages
    if ((is_congested || in_burst) && meta->access_count >= 2 && 
        age_ns < (working_set_window * 2))
        return CACHE_EXT_CONTINUE_ITER;

    // Everything else → evict
    return CACHE_EXT_EVICT_NODE;
}
