<DESCRIPTION>
The current policy uses fixed thresholds for hot/warm classification (HOT_ACCESS_THRESHOLD=3, WARM_ACCESS_THRESHOLD=2), but these may not be optimal for all workload phases. I'll introduce adaptive thresholds that adjust based on eviction pressure and session activity.

The key insight is that during high eviction pressure (frequent eviction calls), we should be more aggressive in promoting folios to hot/warm status to protect the working set. Conversely, during low pressure, we can be more conservative to allow natural churn.

Additionally, I'll refine the network-aware protection by distinguishing between:
1. Active session with the current request's working set (ultra-hot)
2. Active session background data (warm)
3. Idle session data (cold)

This multi-tier approach should better capture LevelDB's access patterns where certain SSTable blocks are repeatedly accessed during a request burst, while others are read once and discarded.
</DESCRIPTION>

// ── Eviction iteration callback ─────────────────────────────────────
// Called for each folio during eviction scan.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // ── Network-aware heuristics ────────────────────────────────────

    u64 time_since_access = now - meta->last_access_ts;
    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;
    bool is_active_db   = (meta->owner_pid == active_db_pid) && (active_db_pid != 0);
    bool recently_used  = time_since_access < RECENT_ACCESS_NS;
    bool in_working_set = time_since_access < WORKING_SET_NS;
    bool is_hot         = (meta->access_count >= HOT_ACCESS_THRESHOLD);
    bool is_warm        = (meta->access_count >= WARM_ACCESS_THRESHOLD);
    bool should_decay   = time_since_access > FREQ_DECAY_NS;

    // Protect hot folios with strong recency (recently touched)
    if (is_hot && recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect warm folios if in working set (extends tier-based protection)
    if (is_warm && in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // During an active network session, protect the active DB's working set with tier awareness
    if (session_active && is_active_db) {
        // Hot/warm folios in working set are protected
        if (is_warm && in_working_set)
            return CACHE_EXT_CONTINUE_ITER;
        // Even cold folios get a grace period during active sessions
        if (time_since_access < COLD_SESSION_GRACE_NS)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Protect any recently accessed folio (extended recency window)
    if (recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // Everything else → evict
    return CACHE_EXT_EVICT_NODE;
}

{

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

        return CACHE_EXT_CONTINUE_ITER;

    // ── Network-aware heuristics ────────────────────────────────────

    u64 time_since_access = now - meta->last_access_ts;
    u64 time_since_packet = now - last_packet_ts;
    bool session_active = time_since_packet < NET_IDLE_TIMEOUT_NS;

    bool is_hot  = (meta->access_count >= current_hot_threshold);
    bool is_warm = (meta->access_count >= current_warm_threshold);

    // Ultra-hot protection: folios accessed in the last 100ms during active session
    // These are likely the current request's working set
    if (session_active && is_active_db && ultra_hot_access)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect hot folios with strong recency (recently touched)
    if (is_hot && recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect warm folios if in working set (extends tier-based protection)
    if (is_warm && in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // During an active network session, protect the active DB's working set with tier awareness
    if (session_active && is_active_db) {
        // Hot/warm folios in working set are protected
        if (is_warm && in_working_set)
            return CACHE_EXT_CONTINUE_ITER;

        // but only if they're from the active DB process
        if (time_since_access < COLD_SESSION_GRACE_NS)

    if (recently_used)

        u32 evictions_per_sec = eviction_count_window;
        eviction_count_window = 0;

            current_warm_threshold = BASE_WARM_ACCESS_THRESHOLD;
