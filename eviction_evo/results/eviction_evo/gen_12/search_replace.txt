<DESCRIPTION>
The current program uses rigid boolean rules (protect if hot+recent OR in_working_set OR recently_used, else evict). This is too inflexible and doesn't allow for nuanced decisions between folios that are equally "recent" but differ in other ways.

I propose converting to a scoring-based approach where each folio gets an evictability score based on multiple factors:
1. Age-based score (older = higher score = more evictable)
2. Access frequency penalty (more accesses = lower score)
3. Network pressure adjustment (under pressure = lower scores = protect more)

This is inspired by the best-performing prior program (score 2162.27) which used scoring. However, I'm keeping the current program's adaptive windows feature and network pressure detection.

The key improvement is replacing the all-or-nothing boolean rules with graduated scoring, allowing the system to make more granular eviction decisions. When memory pressure is high, we can evict the "least protected" folios first, rather than treating all protected folios equally.

This should improve both workloads:
- dual_small_vs_large: Better discrimination between frequently-vs-rarely accessed pages
- dual_congested: More responsive to network condition changes via graduated protection levels

// -- Eviction iteration callback ----------------------------------------------
// Called for each folio during eviction scan.
// Uses a scoring-based approach: higher score = more evictable.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict immediately
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network condition detection ------------------------------------------

    // Loss/retransmission indicators
    bool has_loss = (net_lost > 0) || (net_retrans_out > 0);

    // Cwnd pressure: if packets in flight > cwnd, network is bottlenecked
    bool cwnd_pressure = (net_packets_out > 0) &&
                         ((u32)net_packets_out > (net_snd_cwnd / CWND_PRESSURE_RATIO));

    // High jitter: unstable network (only extreme jitter indicates real pressure)
    bool high_jitter = (net_mdev_us > 2000000);  // > 2ms jitter

    // Network is under pressure if any of: loss, cwnd pressure, high jitter
    bool net_under_pressure = has_loss || cwnd_pressure || high_jitter;

    // Burst detection: short inter-arrival time = burst of requests
    u64 inter_arrival = (net_prev_packet_ts > 0 && last_packet_ts > net_prev_packet_ts) ?
                        (last_packet_ts - net_prev_packet_ts) : (~0ULL);
    bool is_burst = (inter_arrival < 10000000ULL);  // < 10ms = burst

    // Session activity
    bool session_active = (now > last_packet_ts) &&
                         ((now - last_packet_ts) < NET_IDLE_TIMEOUT_NS);

    // -- Adaptive time windows ------------------------------------------------
    u64 adapted_recent = get_adaptive_window(BASE_RECENT_NS, net_under_pressure);
    u64 adapted_working_set = get_adaptive_window(BASE_WORKING_SET_NS, net_under_pressure);

    // -- Calculate eviction score (higher = more evictable) -------------------

    u64 age = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u32 score = 0;

    // Factor 1: Age-based score (normalized to 0-100 range)
    if (age > adapted_working_set) {
        // Very old (beyond working set): base score 80
        score += 80;
    } else if (age > (adapted_working_set >> 1)) {
        // Old (beyond half working set): base score 60
        score += 60;
    } else if (age > adapted_recent) {
        // Moderate age (beyond recent window): base score 40
        score += 40;
    } else {
        // Recent (within recent window): base score 20
        score += 20;
    }

    // Factor 2: Frequency penalty (reduce score for frequently accessed folios)
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        // Hot folios: strong penalty (-30)
        score = (score > 30) ? (score - 30) : 0;
    } else if (meta->access_count >= 2) {
        // Warm folios: moderate penalty (-15)
        score = (score > 15) ? (score - 15) : 0;
    }

    // Factor 3: Tier-based bonus for cold folios
    if (meta->access_tier == 0) {
        score += 20;  // Cold tier bonus
    } else if (meta->access_tier == 1) {
        score += 5;   // Warm tier small bonus
    }

    // Factor 4: Network-aware adjustment
    // Under network pressure, reduce scores (protect more)
    if (net_under_pressure) {
        score = (score * 3) >> 2;  // 0.75x multiplier
    }

    // Factor 5: Burst protection
    // During bursts, further reduce scores for recently accessed folios
    if (is_burst && session_active && age < adapted_working_set) {
        score = score >> 1;  // 0.5x multiplier
    }

    // -- Dynamic eviction threshold -------------------------------------------

    u32 evict_threshold = 60;  // Base threshold

    // Adjust threshold based on network conditions
    if (net_under_pressure) {
        evict_threshold = 70;  // Higher threshold = more protective
    } else if (!session_active) {
        evict_threshold = 40;  // Lower threshold = more aggressive when idle
    }

    // During bursts, increase threshold (protect working set)
    if (is_burst && session_active) {
        evict_threshold += 10;
    }

    // -- Final decision -------------------------------------------------------

    // Evict if score exceeds threshold
    if (score >= evict_threshold) {
        return CACHE_EXT_EVICT_NODE;
    }

    return CACHE_EXT_CONTINUE_ITER;
}
