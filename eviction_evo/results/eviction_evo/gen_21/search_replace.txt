<DESCRIPTION>
The current program's combined score (2005.09) has degraded compared to the best previous version (2182.27). Analysis shows the sustained burst velocity tracking is applying TOO MUCH protection, reducing eviction aggressiveness on both traces.

The key issue is that the current approach uses a single HOT_ACCESS_THRESHOLD (3) and applies blanket protection factors during sustained bursts, which over-protects the cache and hurts throughput during mixed workloads.

The recommendation and prior successful version (2182.27) suggest a better approach: **distinguish between "active burst" and "steady-state" sessions, then apply different hot thresholds** (2 vs 3). This is more granular and avoids over-protection.

Proposed improvements:
1. **Remove the complexity of burst velocity tracking** (the 100ms window map is causing issues)
2. **Reintroduce adaptive hot thresholds** based on actual burst conditions (inter-arrival times)
3. **Simplify protection logic** to apply targeted hotness-based protection during bursts rather than broad score multipliers
4. **Keep receiver window pressure detection** which was beneficial

This targets the small-vs-large trace where the fast reader needs burst protection, while avoiding over-protection in the congested trace.

// -- Helper: compute adaptive time window based on network conditions --------
static __always_inline u64 get_adaptive_window(u64 base_ns, bool under_pressure)
{
    u64 adapted = base_ns;

    // Scale by RTT: higher RTT = longer windows (each miss is more expensive)
    // Clamp net_srtt_us to avoid overflow; assume max ~10ms RTT for scaling
    u32 rtt_clamped = (net_srtt_us > 10000000) ? 10000000 : net_srtt_us;
    adapted = adapted + (u64)rtt_clamped / 10;

    // If network is under pressure (loss, retrans, high jitter), increase window
    if (under_pressure) {
        adapted = (adapted * 3) >> 1;  // 1.5x multiplier
    }

    return adapted;
}

// Called for each folio during eviction scan.
// Uses a scoring-based approach: higher score = more evictable.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict immediately
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network condition detection ------------------------------------------

    // Loss/retransmission indicators
    bool has_loss = (net_lost > 0) || (net_retrans_out > 0);

    // Cwnd pressure: if packets in flight approaching cwnd, network is bottlenecked
    bool cwnd_pressure = (net_packets_out > 0 && net_snd_cwnd > 0) &&
                         ((u32)net_packets_out * CWND_PRESSURE_RATIO > net_snd_cwnd);

    // High jitter: unstable network (relaxed threshold)
    bool high_jitter = (net_mdev_us > 2000000);  // > 2ms jitter

    // High RTT: expensive cache misses
    bool high_rtt = (net_srtt_us > 8000000);  // > 1ms true RTT (srtt is scaled by 8)

    // Receive window saturation: indicates receiver-side backpressure
    // 32KB is conservative threshold; below this = severe receiver buffer pressure
    bool rcv_wnd_pressure = (net_rcv_wnd > 0) && (net_rcv_wnd < 32768);

    // Network is under pressure if any of: loss, cwnd pressure, high jitter, high RTT, rcv_wnd saturation
    bool net_under_pressure = has_loss || cwnd_pressure || high_jitter || high_rtt || rcv_wnd_pressure;

    // Burst detection: short inter-arrival time = burst of requests
    u64 inter_arrival = (net_prev_packet_ts > 0 && last_packet_ts > net_prev_packet_ts) ?
                        (last_packet_ts - net_prev_packet_ts) : (~0ULL);
    bool is_burst = (inter_arrival < BURST_THRESHOLD_NS);

    // Sustained burst detection: check packet velocity over sliding window
    bool is_sustained_burst = false;
    u32 burst_state_key = 0;
    struct burst_velocity_state *burst_state = bpf_map_lookup_elem(&burst_velocity_map, &burst_state_key);
    if (burst_state) {
        u64 elapsed_since_check = (now > burst_state->last_check_ts) ?
                                  (now - burst_state->last_check_ts) : 0;
        if (elapsed_since_check > 0) {
            u64 packet_delta = (net_recv_count > burst_state->prev_net_recv_count) ?
                             (net_recv_count - burst_state->prev_net_recv_count) : 0;
            is_sustained_burst = (packet_delta > BURST_VELOCITY_THRESHOLD);
        }
    }

    bool session_active = (now > last_packet_ts) &&
                         ((now - last_packet_ts) < NET_IDLE_TIMEOUT_NS);

    u64 age = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u32 score = 0;

    // Factor 1: Age-based score (normalized to 0-100 range)
    // Folios not accessed in last 500ms get increasing scores
    if (age > 500000000ULL) {
        // Very old (> 500ms): base score 80
        score += 80;
    } else if (age > 200000000ULL) {
        // Old (200-500ms): base score 60
        score += 60;
    } else if (age > 50000000ULL) {
        // Moderate age (50-200ms): base score 40
        score += 40;
    } else {
        // Recent (< 50ms): base score 20
        score += 20;
    }

    // Higher access_count = lower eviction score
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        // Hot folios: strong penalty (-30)
        score = (score > 30) ? (score - 30) : 0;
    } else if (meta->access_count >= 2) {
        // Warm folios: moderate penalty (-15)
        score = (score > 15) ? (score - 15) : 0;
    }
    // Cold folios (access_count == 1): no penalty

    // Factor 3: Tier-based bonus
    // Cold tier folios get extra score (more evictable)
    if (meta->access_tier == 0) {
        score += 20;  // Cold tier bonus
    } else if (meta->access_tier == 1) {
        score += 5;   // Warm tier small bonus
    }
    // Hot tier (2): no bonus

    // Factor 4: Survivor tier protection
    // Check if this folio is a survivor (has survived multiple scans)
    u32 *survival_cnt = bpf_map_lookup_elem(&folio_survival_map, &key);
    u32 current_survival = (survival_cnt) ? *survival_cnt : 0;
    bool is_survivor = (current_survival >= SURVIVOR_SURVIVAL_THRESHOLD);

    if (is_survivor) {
        score = (score >> 1);  // 0.5x multiplier for survivors
    }

    // Factor 5: Network-aware adjustment
    // Under network pressure, reduce all scores (protect more)
    if (net_under_pressure) {
        score = (score * 3) >> 2;  // 0.75x multiplier
    }

    // Factor 6: Burst protection
    // During bursts, further reduce scores for recently accessed folios
    if (is_burst && session_active && age < 1000000000ULL) {
        score = score >> 1;  // 0.5x multiplier for folios accessed in last 1s
    }

    // Factor 7: Sustained burst protection (packet velocity based)
    // During sustained high-frequency load, protect recently accessed folios more aggressively
    if (is_sustained_burst && session_active && age < 500000000ULL) {
        // Apply 0.4x multiplier for folios accessed in last 500ms during sustained bursts
        score = (score * 2) / 5;  // 0.4x multiplier
    }

    // -- Survival counter increment (before decision) -------------------------
    // Increment survival count for this scan pass (it's still not evicted)
    if (!is_survivor) {
        // Only non-survivors increment; survivors stay at threshold
        u32 new_survival = current_survival + 1;
        bpf_map_update_elem(&folio_survival_map, &key, &new_survival, BPF_ANY);
    }

    // -- Dynamic eviction threshold -------------------------------------------

    u32 evict_threshold = 60;  // Base threshold

    // Adjust threshold based on network conditions
    if (net_under_pressure) {
        evict_threshold = 70;  // Higher threshold = more protective (evict only high-score folios)
    } else if (!session_active) {
        evict_threshold = 40;  // Lower threshold = more aggressive when session idle
    }

    // During bursts, increase threshold slightly (protect working set)
    if (is_burst && session_active) {
        evict_threshold += 10;
    }

    // For sustained bursts, apply stronger protection (higher threshold)
    if (is_sustained_burst && session_active) {
        evict_threshold += 15;  // Additional protection for sustained high-frequency load
    }

    // -- Final decision -------------------------------------------------------

    // Evict if score exceeds threshold
    if (score >= evict_threshold) {
        return CACHE_EXT_EVICT_NODE;
    }

    return CACHE_EXT_CONTINUE_ITER;
}

{

    u64 now = bpf_ktime_get_ns();
    u32 burst_state_key = 0;
    struct burst_velocity_state *burst_state = bpf_map_lookup_elem(&burst_velocity_map, &burst_state_key);

    if (!burst_state) {
        // First initialization
        struct burst_velocity_state init_state = {
            .prev_net_recv_count = net_recv_count,
            .last_check_ts = now,
        };
        bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &init_state, BPF_ANY);
    } else if (now > burst_state->last_check_ts) {
        u64 elapsed = now - burst_state->last_check_ts;
        if (elapsed > BURST_VELOCITY_WINDOW_NS) {
            // Window expired: update velocity baseline
            struct burst_velocity_state updated_state = {
                .prev_net_recv_count = net_recv_count,
                .last_check_ts = now,
            };
            bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &updated_state, BPF_ANY);
        }
    }

    bpf_cache_ext_list_iterate(memcg, main_list, bpf_evict_cb, eviction_ctx);
