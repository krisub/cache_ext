<DESCRIPTION>
This redesign fundamentally changes the program structure from a single-list LRU approach to a multi-tier segregated list architecture. Instead of maintaining one list with metadata-based tier tracking, we now use separate physical lists for hot, warm, and cold folios. This structural change provides several benefits:

1. **Spatial locality**: Eviction scans only the cold list first, reducing iteration overhead
2. **Explicit tier management**: Folios physically move between lists based on access patterns
3. **Adaptive promotion/demotion**: Uses bpf_cache_ext_list_iterate_extended with continue_list for efficient tier transitions
4. **Reduced metadata lookups**: Tier information is implicit in list membership
5. **Better cache utilization**: Hot data stays physically separated from cold data during eviction

The data flow is completely restructured:
- New folios enter the cold list
- Accesses promote folios: cold→warm→hot
- Eviction scans cold list first, then warm if needed
- Periodic demotion moves stale hot/warm folios back down

This architectural change should improve throughput by reducing eviction overhead and better protecting the working set through physical segregation rather than metadata-based filtering.
</DESCRIPTION>

// BPF page cache eviction policy with network awareness.
// This file is evolved by ShinkaEvolve. The EVOLVE-BLOCK contains the
// eviction strategy that will be optimized to maximize throughput.
//
// Fixed infrastructure (outside EVOLVE-BLOCK):
//   - Includes, license, map definitions
//   - Network monitoring kprobe (tcp_recvmsg) that tracks active_db_pid
//   - struct_ops registration at the end
//
// What the LLM can evolve (inside EVOLVE-BLOCK):
//   - Constants and thresholds (e.g., time windows, scoring weights)
//   - The eviction callback (bpf_evict_cb): decides per-folio evict vs keep
//   - folio_added: how new folios are classified and tracked
//   - folio_accessed: how re-accesses update metadata
//   - folio_evicted: cleanup on eviction
//   - evict_folios: the top-level eviction orchestration
//   - Any helper functions or additional data structures
//
// Available BPF APIs (from cache_ext_lib.bpf.h):
//   bpf_cache_ext_list_add(list, folio)        — add folio to head of list
//   bpf_cache_ext_list_add_tail(list, folio)    — add folio to tail of list
//   bpf_cache_ext_list_del(list, folio)         — remove folio from list
//   bpf_cache_ext_list_move(list, folio, tail)  — move folio within list (tail=false→head/front, tail=true→tail/back)
//   bpf_cache_ext_list_iterate(memcg, list, callback, eviction_ctx) — iterate list, callback returns EVICT_NODE or CONTINUE_ITER
//   bpf_cache_ext_list_iterate_extended(memcg, list, callback, opts, eviction_ctx) — iterate with continue_list for promotions
//   bpf_cache_ext_list_sample(memcg, list, callback, ratio, eviction_ctx) — sample random folios
//   bpf_cache_ext_ds_registry_new_list(memcg)   — allocate a new list, returns list handle
//
// Folio helpers:
//   folio_nr_pages(folio), folio_index(folio)
//   folio_test_uptodate/lru/dirty/writeback/unevictable(folio)
//
// Callback return values:
//   CACHE_EXT_EVICT_NODE    — evict this folio
//   CACHE_EXT_CONTINUE_ITER — keep this folio (skip)
//
// Network metrics available as globals:
//   active_db_pid  — PID of process currently handling TCP requests (0 if idle)
//   last_packet_ts — timestamp (ns) of last TCP packet on monitored port
//
// Per-folio metadata in folio_metadata_map:
//   owner_pid      — PID that first brought this folio into cache
//   last_access_ts — timestamp (ns) of last access to this folio

#include "vmlinux.h"
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>
#include <bpf/bpf_core_read.h>
#include "cache_ext_lib.bpf.h"
#include "dir_watcher.bpf.h"

char _license[] SEC("license") = "GPL";

// ── Per-folio metadata ──────────────────────────────────────────────
struct net_folio_metadata {
    u32 owner_pid;
    u64 last_access_ts;
    u32 access_count;        // number of times this folio was accessed
    u32 access_tier;         // tier classification (0=cold, 1=warm, 2=hot)
};

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __type(key, u64);          // (u64)folio pointer
    __type(value, struct net_folio_metadata);
    __uint(max_entries, 1000000);
} folio_metadata_map SEC(".maps");

// ── Debug counters ──────────────────────────────────────────────────
struct {
    __uint(type, BPF_MAP_TYPE_ARRAY);
    __uint(max_entries, 16);
    __type(key, u32);
    __type(value, u64);
} debug_counters SEC(".maps");

// ── Network state (updated by kprobe below) ─────────────────────────
u64 last_packet_ts = 0;
u32 active_db_pid = 0;

// ── Helpers (fixed) ──────────────────────────────────────────────────
static __always_inline void bump_counter(u32 idx) {
    u64 *val = bpf_map_lookup_elem(&debug_counters, &idx);
    if (val) __sync_fetch_and_add(val, 1);
}

static __always_inline bool is_folio_relevant(struct folio *folio) {
    if (!folio || !folio->mapping || !folio->mapping->host)
        return false;
    return inode_in_watchlist(folio->mapping->host->i_ino);
}

// EVOLVE-BLOCK-START
// ── Multi-tier list handles (evolved) ────────────────────────────────
static u64 hot_list;
static u64 warm_list;
static u64 cold_list;

// ══════════════════════════════════════════════════════════════════════
// MULTI-TIER SEGREGATED EVICTION POLICY
// ══════════════════════════════════════════════════════════════════════

// ── Tunable constants ───────────────────────────────────────────────
#define RECENT_ACCESS_NS      300000000ULL   // 300 ms - recent access window
#define NET_IDLE_TIMEOUT_NS   5000000000ULL  // 5 s - network session timeout
#define WORKING_SET_NS        3000000000ULL  // 3 s - working set protection
#define HOT_THRESHOLD         4              // accesses to become hot
#define WARM_THRESHOLD        2              // accesses to become warm
#define DEMOTION_TIMEOUT_NS   3000000000ULL  // 3 s - demote if not accessed
#define COLD_GRACE_NS         150000000ULL   // 150 ms - grace for cold during session

// ── Helper: Determine target tier for a folio ──────────────────────
static __always_inline u32 compute_target_tier(struct net_folio_metadata *meta, u64 now)
{
    if (!meta)
        return 0;
    
    if (meta->access_count >= HOT_THRESHOLD)
        return 2;  // hot
    else if (meta->access_count >= WARM_THRESHOLD)
        return 1;  // warm
    else
        return 0;  // cold
}

// ── Helper: Move folio between tier lists ──────────────────────────
static __always_inline void move_to_tier(struct folio *folio, u32 old_tier, u32 new_tier)
{
    if (old_tier == new_tier)
        return;
    
    // Remove from old list
    bpf_cache_ext_list_del(folio);
    
    // Add to new list (at head for MRU)
    if (new_tier == 2)
        bpf_cache_ext_list_add(hot_list, folio);
    else if (new_tier == 1)
        bpf_cache_ext_list_add(warm_list, folio);
    else
        bpf_cache_ext_list_add(cold_list, folio);
}

// ── Eviction callback for cold list ─────────────────────────────────
static int evict_cold_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // Skip dirty/writeback
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // No metadata → evict
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    u64 time_since_access = now - meta->last_access_ts;
    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;
    bool is_active_db = (meta->owner_pid == active_db_pid) && (active_db_pid != 0);
    bool recently_used = time_since_access < RECENT_ACCESS_NS;

    // Protect recently accessed cold folios
    if (recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // During active session, give cold folios from active DB a grace period
    if (session_active && is_active_db && time_since_access < COLD_GRACE_NS)
        return CACHE_EXT_CONTINUE_ITER;

    // Evict old cold folios
    return CACHE_EXT_EVICT_NODE;
}

// ── Eviction callback for warm list ─────────────────────────────────
static int evict_warm_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // Skip dirty/writeback
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    u64 time_since_access = now - meta->last_access_ts;
    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;
    bool is_active_db = (meta->owner_pid == active_db_pid) && (active_db_pid != 0);
    bool in_working_set = time_since_access < WORKING_SET_NS;

    // Protect warm folios in working set
    if (in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // During active session, protect active DB's warm folios more aggressively
    if (session_active && is_active_db && time_since_access < (WORKING_SET_NS * 2))
        return CACHE_EXT_CONTINUE_ITER;

    // Evict stale warm folios
    return CACHE_EXT_EVICT_NODE;
}

// ── Demotion callback: move stale hot/warm folios down ──────────────
static int demote_hot_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_CONTINUE_ITER;

    u64 time_since_access = now - meta->last_access_ts;

    // If not accessed for DEMOTION_TIMEOUT, demote
    if (time_since_access > DEMOTION_TIMEOUT_NS) {
        // Decrement access count (frequency decay)
        if (meta->access_count > 0)
            meta->access_count -= 1;
        
        u32 new_tier = compute_target_tier(meta, now);
        if (new_tier < 2) {  // demote from hot
            meta->access_tier = new_tier;
            move_to_tier(node->folio, 2, new_tier);
            bump_counter(10);
        }
    }

    return CACHE_EXT_CONTINUE_ITER;
}

static int demote_warm_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_CONTINUE_ITER;

    u64 time_since_access = now - meta->last_access_ts;

    // If not accessed for DEMOTION_TIMEOUT, demote to cold
    if (time_since_access > DEMOTION_TIMEOUT_NS) {
        if (meta->access_count > 0)
            meta->access_count -= 1;
        
        u32 new_tier = compute_target_tier(meta, now);
        if (new_tier < 1) {  // demote from warm to cold
            meta->access_tier = new_tier;
            move_to_tier(node->folio, 1, new_tier);
            bump_counter(11);
        }
    }

    return CACHE_EXT_CONTINUE_ITER;
}

// ── struct_ops: init ────────────────────────────────────────────────
s32 BPF_STRUCT_OPS_SLEEPABLE(evolved_init, struct mem_cgroup *memcg)
{
    hot_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!hot_list)
        return -1;
    
    warm_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!warm_list)
        return -1;
    
    cold_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!cold_list)
        return -1;
    
    return 0;
}

// ── struct_ops: evict_folios ────────────────────────────────────────
void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    bump_counter(0);
    
    // First, try demoting stale hot/warm folios (periodic maintenance)
    // This is lightweight and helps keep tiers accurate
    bpf_cache_ext_list_sample(memcg, hot_list, demote_hot_cb, 10, eviction_ctx);
    bpf_cache_ext_list_sample(memcg, warm_list, demote_warm_cb, 10, eviction_ctx);
    
    // Evict from cold list first (most likely candidates)
    bpf_cache_ext_list_iterate(memcg, cold_list, evict_cold_cb, eviction_ctx);
    
    // If still need to evict, scan warm list
    bpf_cache_ext_list_iterate(memcg, warm_list, evict_warm_cb, eviction_ctx);
    
    // Hot list is protected - only evict if absolutely necessary
    // (kernel will handle this as last resort)
}

// ── struct_ops: folio_added ─────────────────────────────────────────
void BPF_STRUCT_OPS(evolved_folio_added, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    // New folios start in cold list (at tail for FIFO within tier)
    int ret = bpf_cache_ext_list_add_tail(cold_list, folio);
    if (ret != 0) {
        bump_counter(1);
        return;
    }

    // Initialize metadata
    u64 key = (u64)folio;
    struct net_folio_metadata new_meta = {
        .owner_pid     = bpf_get_current_pid_tgid() >> 32,
        .last_access_ts = bpf_ktime_get_ns(),
        .access_count  = 1,
        .access_tier   = 0,  // cold
    };
    bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
    bump_counter(2);
}

// ── struct_ops: folio_accessed ──────────────────────────────────────
void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)folio;
    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta) {
        // Folio not tracked - add to cold list
        int ret = bpf_cache_ext_list_add(cold_list, folio);
        if (ret != 0) {
            // Already in a list - move to head within current tier
            bpf_cache_ext_list_move(cold_list, folio, false);
        }
        struct net_folio_metadata new_meta = {
            .owner_pid     = bpf_get_current_pid_tgid() >> 32,
            .last_access_ts = now,
            .access_count  = 1,
            .access_tier   = 0,
        };
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    // Update metadata
    u32 old_tier = meta->access_tier;
    meta->last_access_ts = now;
    meta->access_count += 1;

    // Compute new tier based on access count
    u32 new_tier = compute_target_tier(meta, now);
    meta->access_tier = new_tier;

    // Promote/demote between lists if tier changed
    if (new_tier != old_tier) {
        move_to_tier(folio, old_tier, new_tier);
        bump_counter(4);
    } else {
        // Same tier - just move to head (MRU within tier)
        if (new_tier == 2)
            bpf_cache_ext_list_move(hot_list, folio, false);
        else if (new_tier == 1)
            bpf_cache_ext_list_move(warm_list, folio, false);
        else
            bpf_cache_ext_list_move(cold_list, folio, false);
        bump_counter(5);
    }
}

// ── struct_ops: folio_evicted ───────────────────────────────────────
void BPF_STRUCT_OPS(evolved_folio_evicted, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    // Remove from whichever list it's in
    bpf_cache_ext_list_del(folio);

    u64 key = (u64)folio;
    bpf_map_delete_elem(&folio_metadata_map, &key);
    bump_counter(6);
}

// EVOLVE-BLOCK-END

// ══════════════════════════════════════════════════════════════════════
// NETWORK MONITORING (fixed — not evolved)
// ══════════════════════════════════════════════════════════════════════

SEC("kprobe/tcp_recvmsg")
int trace_tcp_recvmsg(struct pt_regs *ctx)
{
    struct sock *sk = (struct sock *)PT_REGS_PARM1(ctx);
    if (!sk) return 0;

    u16 sport = BPF_CORE_READ(sk, __sk_common.skc_num);

    // Monitor the net_leveldb_server port (9100) and common DB ports
    if (sport == 9100 || sport == 9001 || sport == 9002) {
        last_packet_ts = bpf_ktime_get_ns();
        active_db_pid  = bpf_get_current_pid_tgid() >> 32;
    }
    return 0;
}

    .init           = (void *)evolved_init,
    .evict_folios   = (void *)evolved_evict_folios,
    .folio_added    = (void *)evolved_folio_added,
    .folio_evicted  = (void *)evolved_folio_evicted,
    .folio_accessed = (void *)evolved_folio_accessed,
    .admit_folio    = (void *)0,
};
