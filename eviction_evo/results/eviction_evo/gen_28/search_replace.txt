<DESCRIPTION>
The current program with burst velocity tracking has regressed (1946.20 score) compared to the best prior implementation (2182.27 score). The burst velocity mechanism is too coarse and over-protects the cache, reducing eviction efficiency on the small_vs_large trace.

The recommendation suggests implementing a "working set stability score" that samples folios within the adapted working set window to calibrate the hot threshold dynamically. This data-driven approach will:

1. Sample up to 50 folios within the adapted_working_set time window during each eviction scan
2. Count how many were accessed within the last 100ms
3. If >80% of sampled folios show recent access, lower the dynamic_hot_threshold by 1 (making more folios eligible for eviction), indicating a stable, frequently-accessed working set where we can afford tighter eviction
4. This replaces the coarse burst velocity mechanism with a precise, locality-aware calibration

This approach adapts thresholds based on actual working set characteristics rather than raw packet counts, which should improve both traces:
- For small_vs_large: the small reader's hot working set will be detected and protected appropriately
- For congested: the data-driven approach adapts independently of network conditions, providing robustness

// -- Helper: compute adaptive time window based on network conditions --------
static __always_inline u64 get_adaptive_window(u64 base_ns, bool under_pressure)
{
    u64 adapted = base_ns;

    // Scale by RTT: higher RTT = longer windows (each miss is more expensive)
    // Clamp net_srtt_us to avoid overflow; assume max ~10ms RTT for scaling
    u32 rtt_clamped = (net_srtt_us > 10000000) ? 10000000 : net_srtt_us;
    adapted = adapted + (u64)rtt_clamped / 10;

    // If network is under pressure (loss, retrans, high jitter), increase window
    if (under_pressure) {
        adapted = (adapted * 3) >> 1;  // 1.5x multiplier
    }

    return adapted;
}

{

        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict immediately
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network condition detection ------------------------------------------

    // Loss/retransmission indicators
    bool has_loss = (net_lost > 0) || (net_retrans_out > 0);

    // Cwnd pressure: if packets in flight approaching cwnd, network is bottlenecked
    bool cwnd_pressure = (net_packets_out > 0 && net_snd_cwnd > 0) &&
                         ((u32)net_packets_out * CWND_PRESSURE_RATIO > net_snd_cwnd);

    // High jitter: unstable network (relaxed threshold)
    bool high_jitter = (net_mdev_us > 2000000);  // > 2ms jitter

    // High RTT: expensive cache misses
    bool high_rtt = (net_srtt_us > 8000000);  // > 1ms true RTT (srtt is scaled by 8)

    // Receive window saturation: indicates receiver-side backpressure
    // 32KB is conservative threshold; below this = severe receiver buffer pressure
    bool rcv_wnd_pressure = (net_rcv_wnd > 0) && (net_rcv_wnd < 32768);

    // Network is under pressure if any of: loss, cwnd pressure, high jitter, high RTT, rcv_wnd saturation
    bool net_under_pressure = has_loss || cwnd_pressure || high_jitter || high_rtt || rcv_wnd_pressure;

    // Session activity
    bool session_active = (now > last_packet_ts) &&
                         ((now - last_packet_ts) < NET_IDLE_TIMEOUT_NS);

// -- Eviction iteration callback ----------------------------------------------
// Called for each folio during eviction scan.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network condition detection ------------------------------------------

    // Loss/retransmission indicators
    bool has_loss = (net_lost > 0) || (net_retrans_out > 0);

    // Cwnd pressure: if packets in flight > cwnd, network is bottlenecked
    bool cwnd_pressure = (net_packets_out > 0) &&
                         ((u32)net_packets_out > (net_snd_cwnd / CWND_PRESSURE_RATIO));

    // Adaptive jitter threshold based on RTT percentile
    // Scale 2ms baseline by (RTT / 100ms), clamped to [0.5x, 2.0x] → [1ms, 4ms]
    u32 jitter_threshold_us = 2000000;  // baseline 2ms
    if (net_srtt_us > 0) {
        // Calculate threshold = 2ms * (net_srtt_us / 100000us)
        // Clamp scaling ratio to [0.5, 2.0] to avoid extreme thresholds
        u64 scaled = ((u64)net_srtt_us * 2000000) / 100000;
        if (scaled < 1000000) scaled = 1000000;    // 0.5x minimum → 1ms
        if (scaled > 4000000) scaled = 4000000;    // 2.0x maximum → 4ms
        jitter_threshold_us = (u32)scaled;
    }
    bool high_jitter = (net_mdev_us > jitter_threshold_us);

    // Receive window saturation: indicates receiver-side backpressure
    // 64KB is a typical minimal buffer; below this = receiver can't accept more data
    bool rcv_wnd_pressure = (net_rcv_wnd > 0) && (net_rcv_wnd < 65536);

    // Network is under pressure if any of: loss, cwnd pressure, high jitter, rcv_wnd saturation
    bool net_under_pressure = has_loss || cwnd_pressure || high_jitter || rcv_wnd_pressure;

    // Session activity (check early for threshold adaptation)
    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;

    // -- Adaptive time windows ------------------------------------------------
    u64 adapted_recent = get_adaptive_window(BASE_RECENT_NS, net_under_pressure);
    u64 adapted_working_set = get_adaptive_window(BASE_WORKING_SET_NS, net_under_pressure);

    // -- Dynamic hot threshold based on network conditions -------------------
    u32 dynamic_hot_threshold = get_hot_access_threshold(net_under_pressure, session_active);

    // -- Network-aware heuristics ---------------------------------------------

    bool recently_used  = (now - meta->last_access_ts) < adapted_recent;
    bool in_working_set = (now - meta->last_access_ts) < adapted_working_set;
    bool is_hot         = (meta->access_count >= dynamic_hot_threshold);

    // Protect hot folios (many accesses) that were recently touched
    if (is_hot && recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // During an active network session, protect the working set
    if (session_active && in_working_set)
        return CACHE_EXT_CONTINUE_ITER;

    // Protect any recently accessed folio
    if (recently_used)
        return CACHE_EXT_CONTINUE_ITER;

    // Everything else → evict
    return CACHE_EXT_EVICT_NODE;
}

void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    bump_counter(0);

    // Update burst velocity tracking state before eviction scan
    u64 now = bpf_ktime_get_ns();
    u32 burst_state_key = 0;
    struct burst_velocity_state *burst_state = bpf_map_lookup_elem(&burst_velocity_map, &burst_state_key);

    if (!burst_state) {
        // First initialization
        struct burst_velocity_state init_state = {
            .prev_net_recv_count = net_recv_count,
            .last_check_ts = now,
            .burst_packet_count = 0,
        };
        bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &init_state, BPF_ANY);
    } else if (now > burst_state->last_check_ts) {
        u64 elapsed = now - burst_state->last_check_ts;
        if (elapsed > BURST_VELOCITY_WINDOW_NS) {
            // Window expired: reset velocity baseline
            struct burst_velocity_state updated_state = {
                .prev_net_recv_count = net_recv_count,
                .last_check_ts = now,
                .burst_packet_count = 0,
            };
            bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &updated_state, BPF_ANY);
        }
    }

    bpf_cache_ext_list_iterate(memcg, main_list, bpf_evict_cb, eviction_ctx);
}
