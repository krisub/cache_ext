<DESCRIPTION>
The current program regressed from a prior best score of 1977.84 to 1823.69, primarily due to an overly aggressive recency decay factor that penalizes older pages too quickly. The best-performing version used simple, fixed interval thresholds (< 1s for hot, > 10s for cold) with minimal complexity.

The recommendation suggests computing workload-adaptive thresholds based on interval_ewma statistics collected during eviction cycles. This approach:

1. **Adds adaptive thresholds**: Instead of fixed 1s/10s cutoffs, compute them dynamically from observed per-folio access intervals. This avoids either over-protecting or under-protecting the working set depending on workload characteristics.

2. **Simplifies decay**: Replace the aggressive linear decay with a gentler approach that primarily uses interval EWMA to distinguish hot from cold, avoiding the complex time-based penalty.

3. **Removes complexity**: The current decay_factor calculation is overly aggressive for pages that haven't been accessed in a few seconds but are still part of the working set. By relying on interval metrics (which naturally capture access patterns), we avoid penalizing legitimate variations in access timing.

4. **Collects distribution statistics**: Maintain a small map tracking min/max/median of observed interval_ewma values to calibrate thresholds dynamically, making the policy adaptive to workload patterns (sparse vs. bursty).

The strategy combines the simplicity of the prior best version with workload awareness, expected to yield improvements across both traces by better distinguishing genuinely hot folios from moderately-accessed ones.

/*
 * Enhanced LFU with recency decay and network-aware protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Access count is normalized by age (decay) to avoid bias toward long-lived pages.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;

    /* Base score with recency decay: penalize pages that haven't been accessed recently.
     * Over 5 seconds, decay access_count to 50%. After 10 seconds, to ~0%.
     * Decay factor: 1.0 at now, decays toward 0 as time_since_access increases.
     */
    s64 base_count = (s64)meta->access_count;
    s64 decay_factor = 1000;  /* fixed-point: 1.0 = 1000 */

    if (time_since_access > 0 && time_since_access < 10000000000ULL) {  /* 10 seconds */
        decay_factor = 1000 - (time_since_access / 10000000);  /* Linear decay from 1000 to 0 over 10s */
        if (decay_factor < 10) decay_factor = 10;  /* Floor at small positive */
    } else if (time_since_access >= 10000000000ULL) {
        decay_factor = 10;  /* Very old pages get minimal score */
    }

    s64 base_score = (base_count * decay_factor) / 1000;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second interval */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds interval */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.

{

    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: use interval EWMA with adaptive thresholds based on workload.
     * Compute dynamic hot/cold thresholds from observed interval_ewma statistics.
     * Default fallback if no statistics yet: hot if < 1s, cold if > 10s.
     */
    s64 interval_ewma = meta->interval_ewma.value;
    
    u32 stats_key = 0;
    struct {
        u64 interval_min;
        u64 interval_max;
        u64 interval_sum;
        u32 interval_count;
        u32 eviction_cycles;
    } *stats = bpf_map_lookup_elem(&interval_stats_map, &stats_key);
    
    s64 hot_threshold = 1000000000;      /* Default: < 1 second */
    s64 cold_threshold = 10000000000;    /* Default: > 10 seconds */
    
    if (stats && stats->interval_count > 10) {
        /* Compute adaptive thresholds from observed distribution */
        s64 mean_interval = stats->interval_sum / stats->interval_count;
        /* Hot threshold: 0.5x median/mean (more frequent than average) */
        hot_threshold = (mean_interval / 2);
        if (hot_threshold < 500000000) hot_threshold = 500000000;  /* Floor at 0.5s */
        /* Cold threshold: 5x median/mean (less frequent than average) */
        cold_threshold = (mean_interval * 5);
        if (cold_threshold > 15000000000ULL) cold_threshold = 15000000000ULL;  /* Cap at 15s */
    }
    
    if (interval_ewma > 0 && interval_ewma < hot_threshold) {
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > cold_threshold) {
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;

void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta) {
        int ret = bpf_cache_ext_list_add(main_list, folio);
        if (ret != 0)
            bpf_cache_ext_list_move(main_list, folio, false);
        struct vulcan_folio_metadata new_meta =
            vulcan_folio_init(bpf_ktime_get_ns());
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    vulcan_folio_on_access(meta, bpf_ktime_get_ns(), &folio_cfg);
    
    /* Collect interval statistics during first 100 eviction cycles */
    u32 stats_key = 0;
    struct {
        u64 interval_min;
        u64 interval_max;
        u64 interval_sum;
        u32 interval_count;
        u32 eviction_cycles;
    } *stats = bpf_map_lookup_elem(&interval_stats_map, &stats_key);
    
    if (stats && stats->eviction_cycles < 100) {
        s64 interval_val = meta->interval_ewma.value;
        if (interval_val > 0) {
            if (stats->interval_count == 0) {
                stats->interval_min = interval_val;
                stats->interval_max = interval_val;
            } else {
                if ((u64)interval_val < stats->interval_min)
                    stats->interval_min = interval_val;
                if ((u64)interval_val > stats->interval_max)
                    stats->interval_max = interval_val;
            }
            stats->interval_sum += interval_val;
            stats->interval_count += 1;
        }
    }
    
    bump_counter(4);
}
