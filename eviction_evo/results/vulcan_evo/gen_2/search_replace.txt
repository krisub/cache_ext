<DESCRIPTION>
The baseline pure LFU policy ignores network conditions, which is suboptimal for network-bound workloads. This change introduces network-adaptive eviction by:

1. **Enabling RTT monitoring** (GF_SRTT_US) with EWMA smoothing to track network latency. High RTT means each cache miss is expensive, so we should protect frequently accessed pages more aggressively.

2. **Enabling loss detection** (GF_LOST) with MinMax to identify congestion events. When packet loss occurs, we become more conservative about eviction.

3. **Enabling congestion window tracking** (GF_SND_CWND) with EWMA to understand network capacity utilization.

4. **Enabling inter-arrival time tracking** (GF_INTER_ARRIVAL) with rolling window to detect request bursts and adapt protection accordingly.

5. **Enabling per-folio interval tracking** to distinguish genuinely hot pages (low inter-access interval) from pages with high access counts but infrequent recent activity.

The scoring function is modified to:
- Use a baseline from access_count (frequency component)
- Apply RTT-based multipliers: when RTT is elevated, multiply the protection score to keep hot pages longer
- Penalize eviction when loss is detected (add penalty to make pages less attractive as victims)
- Use per-folio interval EWMA: pages with low intervals (hot) get additional protection
- Detect bursts via inter-arrival and increase protection during high request rates

This approach adapts eviction decisions to real-time network conditions while maintaining frequency-based foundation.

/*
 * Baseline: pure LFU (mimics cache_ext_sampling.bpf.c bpf_lfu_score_fn).
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Evolution can add vulcan_get_* calls and protection rules.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 score = (s64)meta->access_count;

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Network-adaptive scoring: adjust eviction based on RTT, loss, and per-folio heat.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        base_score += 100000;

    // Network-adaptive adjustments
    s64 rtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 rtt_min  = vulcan_get_min(GF_SRTT_US);
    s64 lost_max = vulcan_get_max(GF_LOST);
    s64 cwnd_ewma = vulcan_get_ewma(GF_SND_CWND);
    s64 inter_arrival_avg = vulcan_get_window_avg(GF_INTER_ARRIVAL);

    // RTT pressure: when RTT is elevated, protect hot pages more
    // If rtt_ewma > rtt_min * 1.5, apply a multiplier
    if (rtt_min > 0 && rtt_ewma > 0) {
        // Scale: if RTT doubled, multiply score by 2x
        s64 rtt_ratio = (rtt_ewma * 100) / (rtt_min + 1);
        if (rtt_ratio > 150) {  // RTT elevated by 50%+
            s64 multiplier = (rtt_ratio - 100) / 50;  // +1x for each 50% increase
            if (multiplier > 5) multiplier = 5;  // cap at 5x
            base_score += base_score * multiplier;
        }
    }

    // Loss penalty: if we've seen packet loss, be more conservative
    if (lost_max > 0) {
        base_score += 5000;  // Add penalty to protect more pages
    }

    // Per-folio heat: lower interval EWMA = hotter page
    // Protect pages with low inter-access intervals
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 50000000) {  // < 50ms = hot
        s64 heat_bonus = (50000000 - interval_ewma) / 10000;
        base_score += heat_bonus;
    }

    // Burst detection: low inter-arrival = high request rate
    // Protect working set during bursts
    if (inter_arrival_avg > 0 && inter_arrival_avg < 100000) {  // < 100us = burst
        base_score += 2000;
    }

    // Congestion window: if cwnd is low, be conservative
    if (cwnd_ewma > 0 && cwnd_ewma < 10) {
        base_score += 3000;
    }

    return base_score;
}
