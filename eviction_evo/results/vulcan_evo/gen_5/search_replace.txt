<DESCRIPTION>
The current implementation adds many bonuses to the score uniformly, which may not effectively distinguish between truly hot and cold pages. The performance regression on ycsb_a_d (4963→4384) suggests the policy is being too protective overall.

Key improvements:
1. Add MINMAX listener to GF_SRTT_US to detect RTT spikes and normalize against historical values
2. Enable PACKETS_OUT listener to better detect congestion (compare to CWND)
3. Reduce per-folio listener alpha for faster adaptation (200→150)
4. Refine scoring to use relative network conditions rather than absolute thresholds:
   - Use RTT relative to its EWMA (current vs smoothed average)
   - Use cwnd/packets_out ratio for congestion detection
   - Make bonuses more selective - only protect high-access-count pages during adverse network conditions
5. Simplify recency logic - the 30-second window may be too aggressive
6. Keep interval-based heat detection but reduce its bonus to avoid over-protection

This approach should better adapt to different workload patterns by being more selective about which pages to protect, while still being network-aware.
</DESCRIPTION>

static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    // Base score: access count (higher = keep)
    s64 score = (s64)meta->access_count;

    // Network-aware protection: when RTT is high, cache misses are expensive
    // RTT is in usec; typical good RTT < 10ms (10000us), bad > 50ms (50000us)
    s64 rtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    if (rtt_ewma > 50000) {
        // High RTT: add bonus to protect frequently accessed pages
        score += meta->access_count / 2;
    } else if (rtt_ewma > 20000) {
        // Moderate RTT: small bonus
        score += meta->access_count / 4;
    }

    // Loss detection: if we've seen packet loss, be conservative
    s64 lost_max = vulcan_get_max(GF_LOST);
    if (lost_max > 0) {
        // Network congestion: protect working set
        score += meta->access_count / 3;
    }

    // Per-folio heat: lower interval EWMA = hotter (more frequent accesses)
    // Interval is in nanoseconds; < 100ms (100000000ns) = hot
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 100000000) {
        // Hot folio: frequently re-accessed, add bonus
        score += 5000;
    } else if (interval_ewma > 0 && interval_ewma < 1000000000) {
        // Warm folio: < 1 second interval
        score += 2000;
    }

    // Recency: protect recently accessed folios (likely in working set)
    u64 now = bpf_ktime_get_ns();
    s64 age = (s64)(now - meta->last_access_ts);
    if (age < 5000000000) {  // < 5 seconds
        score += 3000;
    } else if (age < 30000000000) {  // < 30 seconds
        score += 1000;
    }

    // LevelDB: protect index block (last page of file)
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    // Base score: access count (higher = keep)
    s64 score = (s64)meta->access_count;

    // Adaptive network-aware protection using relative RTT
    s64 rtt_current = vulcan_get_ewma(GF_SRTT_US);
    s64 rtt_max = vulcan_get_max(GF_SRTT_US);

    // Only protect frequently accessed pages during network stress
    if (meta->access_count >= 3) {
        // RTT spike detection: if current RTT is much higher than typical
        if (rtt_max > 10000 && rtt_current > (rtt_max * 7) / 10) {
            // Current RTT is >= 70% of max seen - network is degraded
            score += meta->access_count / 2;
        }

        // Congestion detection via cwnd vs packets_out
        s64 cwnd = vulcan_get_ewma(GF_SND_CWND);
        s64 pkts_out = vulcan_get_ewma(GF_PACKETS_OUT);
        if (cwnd > 0 && pkts_out > (cwnd * 8) / 10) {
            // Network is congested (packets in flight > 80% of cwnd)
            score += meta->access_count / 3;
        }
    }

    // Loss detection: if we've seen packet loss, protect high-access pages
    s64 lost_max = vulcan_get_max(GF_LOST);
    if (lost_max > 0 && meta->access_count >= 4) {
        score += meta->access_count / 4;
    }

    // Per-folio heat: lower interval EWMA = hotter (more frequent accesses)
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 100000000) {
        // Very hot folio: < 100ms inter-access
        score += 3000;
    } else if (interval_ewma > 0 && interval_ewma < 500000000) {
        // Warm folio: < 500ms inter-access
        score += 1000;
    }

    // Recency: protect recently accessed folios in working set
    u64 now = bpf_ktime_get_ns();
    s64 age = (s64)(now - meta->last_access_ts);
    if (age < 3000000000) {  // < 3 seconds, likely in active working set
        score += 2000;
    } else if (age < 10000000000) {  // < 10 seconds
        score += 500;
    }

    // LevelDB: protect index block (last page of file)
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
