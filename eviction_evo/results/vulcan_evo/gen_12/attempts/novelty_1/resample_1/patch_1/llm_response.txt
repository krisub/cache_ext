<DESCRIPTION>
Analyzing the prior programs, the best performer (1977.84) used simple static RTT thresholds (>10ms=1.5x, >5ms=1.25x) without time-decay. The current program (1823.69) introduced a decay_factor based on time_since_access, which degraded performance.

The recommendation suggests implementing adaptive RTT thresholds via baseline drift detection. The key insight: instead of fixed 5ms/10ms thresholds, establish a long-term RTT baseline (using very low EWMA alpha ~50) and apply adaptive protection multipliers based on deviation from baseline.

This approach:
1. **Removes decay_factor** — it was over-penalizing aged pages, preventing good locality exploitation
2. **Adds baseline RTT tracking** — maintains a slow-moving EWMA (alpha=50) of SRTT to establish "normal" RTT
3. **Scales protection dynamically** — multipliers grow with RTT deviation from baseline, not absolute thresholds
4. **Detects sustained congestion** — deviations > baseline*1.2 indicate degraded conditions requiring stronger working-set protection

This preserves the proven static logic while adapting to workload-specific network baselines, improving robustness across diverse network conditions.

/*
 * Enhanced LFU with recency decay and network-aware protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Access count is normalized by age (decay) to avoid bias toward long-lived pages.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;

    /* Base score with recency decay: penalize pages that haven't been accessed recently.
     * Over 5 seconds, decay access_count to 50%. After 10 seconds, to ~0%.
     * Decay factor: 1.0 at now, decays toward 0 as time_since_access increases.
     */
    s64 base_count = (s64)meta->access_count;
    s64 decay_factor = 1000;  /* fixed-point: 1.0 = 1000 */

    if (time_since_access > 0 && time_since_access < 10000000000ULL) {  /* 10 seconds */
        decay_factor = 1000 - (time_since_access / 10000000);  /* Linear decay from 1000 to 0 over 10s */
        if (decay_factor < 10) decay_factor = 10;  /* Floor at small positive */
    } else if (time_since_access >= 10000000000ULL) {
        decay_factor = 10;  /* Very old pages get minimal score */
    }

    s64 base_score = (base_count * decay_factor) / 1000;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second interval */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds interval */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * LFU with adaptive network-aware protection via RTT baseline drift detection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Protection scales with deviation of current RTT from long-term baseline.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Base score: pure frequency (no decay) — works better than time-decay */
    s64 base_score = (s64)meta->access_count;

    /* Adaptive network-aware protection based on RTT baseline drift */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Compare current RTT to baseline to detect sustained degradation */
    if (srtt_baseline_ewma > 0) {
        s64 baseline = srtt_baseline_ewma;
        s64 current = srtt_ewma;
        
        /* Compute deviation ratio: (current / baseline) * 100 */
        s64 deviation_ratio = (current * 100) / baseline;
        
        /* Scale protection based on deviation from baseline:
         * - 100% (current == baseline): 1.0x (network_factor = 100)
         * - 120% (current == 1.2x baseline): 1.25x (network_factor = 125)
         * - 150% (current == 1.5x baseline): 1.5x (network_factor = 150)
         * - 200%+ (severe degradation): 1.75x (network_factor = 175) */
        if (deviation_ratio > 200) {
            network_factor = 175;  /* Severe degradation */
        } else if (deviation_ratio > 150) {
            network_factor = 150;
        } else if (deviation_ratio > 120) {
            network_factor = 125;
        }
    } else {
        /* Fallback if baseline not yet initialized: use absolute thresholds */
        if (srtt_ewma > 10000) {
            network_factor = 150;
        } else if (srtt_ewma > 5000) {
            network_factor = 125;
        }
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second interval */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds interval */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
