<DESCRIPTION>
The current program uses fixed RTT thresholds (5ms, 10ms) that don't adapt to workload-specific network characteristics. The recommendation suggests implementing adaptive RTT baseline tracking to dynamically calibrate protection multipliers based on long-term network conditions.

**Rationale:** 
- In dual_small_vs_large, RTT varies naturally; fixed thresholds may trigger protection prematurely during normal operation, hurting the fast reader workload (currently 1529.57 MB/s vs best 1601.83 MB/s).
- A long-term baseline EWMA (alpha ~50 for slow adaptation) captures normal RTT, while current SRTT_US listener (alpha 150) reacts to recent variations.
- Dynamic threshold scaling: compare current RTT to baseline, apply proportional protection boost only on significant deviations (>1.2x baseline), preserving normal-case performance while guarding against congestion.
- This extends Generation 14's fixed-threshold approach (score 2090.49) with adaptive calibration to diverse network conditions.

**Implementation:**
1. Add a single-entry map `rtt_baseline_map` to store long-term RTT baseline.
2. Update baseline in kprobe with ultra-low alpha (50) to distinguish signal from noise.
3. Modify `bpf_score_fn` to compute dynamic threshold based on (current_srtt - baseline) ratio.
4. Scale network_factor only when RTT exceeds baseline + dynamic margin, reducing unnecessary protection during normal operation while maintaining robustness under congestion.

/*
 * Recency-weighted LFU with network-adaptive protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 *
 * Key innovation: base_score combines access frequency with recency weighting
 * to distinguish recently-active frequent pages from cold, infrequently-accessed
 * pages. Network conditions (RTT, loss) trigger adaptive protection to prevent
 * thrashing during congestion.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Recency-weighted access count: boost recently-accessed pages.
     * Linear decay from 80% boost (< 1 second) to 0% boost (10+ seconds).
     * Formula: recency_boost = 80 * max(0, 1 - time_since_access / 10s)
     * Applied as: base_score = access_count * (100 + recency_boost) / 100 */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    s64 recency_boost = 0;

    if (time_since_access < 10000000000ULL) {  /* Within 10 seconds */
        s64 elapsed_deciseconds = time_since_access / 1000000000ULL;  /* Convert ns to deciseconds (0.1s units) */
        recency_boost = 80 - (elapsed_deciseconds * 8);  /* Linear decay: 80 at 0s, 0 at 10s */
        if (recency_boost < 0) recency_boost = 0;
    }

    s64 base_score = ((s64)meta->access_count * (100 + recency_boost)) / 100;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_ewma > 3000) {  /* Loss + RTT > 3ms */
        network_factor = (network_factor * 115) / 100;  /* additional 15% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios.
     * Low interval_ewma = frequent accesses (hot).
     * High interval_ewma = infrequent accesses (cold). */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Adaptive-RTT LFU with dynamic threshold scaling.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 *
 * Key innovation: protection multipliers scale dynamically based on deviation
 * from long-term RTT baseline, not fixed thresholds. This distinguishes normal
 * RTT variation from genuine congestion, preserving performance during normal
 * operation while guarding against network degradation.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Network-aware protection with adaptive RTT baseline scaling */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);

    /* Fetch long-term RTT baseline (ultra-low alpha for stability) */
    u32 baseline_key = 0;
    struct vulcan_ewma *baseline_ewma = bpf_map_lookup_elem(&rtt_baseline_map, &baseline_key);
    s64 rtt_baseline = (baseline_ewma && baseline_ewma->value > 0) ? baseline_ewma->value : srtt_ewma;

    /* Dynamic threshold scaling: compute protection based on deviation from baseline.
     * Protection kicks in only when RTT exceeds baseline by a meaningful margin (>20%).
     * Formula: if current > baseline * 1.2, apply scaling proportional to deviation.
     * - At baseline * 1.2 (20% above): 1.15x protection
     * - At baseline * 1.5 (50% above): 1.4x protection
     * - At baseline * 2.0 (100% above): 1.65x protection
     */
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    if (srtt_ewma > rtt_baseline && rtt_baseline > 0) {
        s64 deviation_ratio = (srtt_ewma * 100) / rtt_baseline;  /* e.g., 150 = 1.5x baseline */

        if (deviation_ratio > 150) {
            /* >50% above baseline: strong protection (1.65x) */
            network_factor = 165;
        } else if (deviation_ratio > 140) {
            /* 40-50% above baseline: moderate protection (1.4x) */
            network_factor = 140;
        } else if (deviation_ratio > 130) {
            /* 30-40% above baseline: light protection (1.25x) */
            network_factor = 125;
        } else if (deviation_ratio > 120) {
            /* 20-30% above baseline: minimal protection (1.15x) */
            network_factor = 115;
        }
    }

    /* Packet loss detected: additional conservative boost when loss + elevated RTT */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_ewma > 3000) {  /* Loss + RTT > 3ms */
        network_factor = (network_factor * 115) / 100;  /* additional 15% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: protect pages accessed recently */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 5000000000ULL) {  /* 5 seconds in ns */
        score += 50;
    }

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios.
     * Low interval_ewma = frequent accesses (hot).
     * High interval_ewma = infrequent accesses (cold). */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
