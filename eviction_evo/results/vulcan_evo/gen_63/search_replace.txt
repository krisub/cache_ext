<DESCRIPTION>
The current program added adaptive interval threshold complexity that introduced overhead without improving performance. The best-performing Generation 39 used a simpler approach with fine-grained recency weighting. This edit:

1. Removes the interval_stats_map and all adaptive threshold logic - this complexity degraded performance
2. Restores Generation 39's proven fine-grained recency-weighted base score calculation with multi-tier decay (50% boost at <1s, linear decay to 0% at 10s)
3. Simplifies the per-folio heat logic back to fixed thresholds that worked well in Generation 39
4. Keeps the proven network-aware protection mechanisms (RTT EWMA, loss detection)

The key insight from the recommendation is that the recency-weighted access count formula from Generation 39 (`(access_count * (100 + recency_factor)) / 100`) already provides good normalization by making temporal locality a multiplicative component of the base score. The current program's regression came from abandoning this proven approach in favor of untested adaptive complexity.

/*
 * Simplified LFU with workload-adaptive interval thresholds and network-aware protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Thresholds adapt based on observed per-folio interval_ewma distribution.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: use interval EWMA with adaptive thresholds based on workload.
     * Compute dynamic hot/cold thresholds from observed interval_ewma statistics.
     * Default fallback if no statistics yet: hot if < 1s, cold if > 10s.
     */
    s64 interval_ewma = meta->interval_ewma.value;

    u32 stats_key = 0;
    struct {
        u64 interval_min;
        u64 interval_max;
        u64 interval_sum;
        u32 interval_count;
        u32 eviction_cycles;
    } *stats = bpf_map_lookup_elem(&interval_stats_map, &stats_key);

    s64 hot_threshold = 1000000000;      /* Default: < 1 second */
    s64 cold_threshold = 10000000000;    /* Default: > 10 seconds */

    if (stats && stats->interval_count > 10) {
        /* Compute adaptive thresholds from observed distribution */
        s64 mean_interval = stats->interval_sum / stats->interval_count;
        /* Hot threshold: 0.5x median/mean (more frequent than average) */
        hot_threshold = (mean_interval / 2);
        if (hot_threshold < 500000000) hot_threshold = 500000000;  /* Floor at 0.5s */
        /* Cold threshold: 5x median/mean (less frequent than average) */
        cold_threshold = (mean_interval * 5);
        if (cold_threshold > 15000000000ULL) cold_threshold = 15000000000ULL;  /* Cap at 15s */
    }

    if (interval_ewma > 0 && interval_ewma < hot_threshold) {
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > cold_threshold) {
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Network-aware LFU with fine-grained recency weighting and per-folio heat tracking.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 *
 * Combines RTT sensitivity, per-folio heat tracking (via interval_ewma),
 * and multi-tier recency decay to make intelligent eviction decisions.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Recency-weighted base score: prioritize recently-active folios over cold ones
     * with similar historical access counts. Linear decay from +50% boost (< 1 second)
     * to 0% boost (10+ seconds). This captures working-set dynamics by making temporal
     * locality a first-class component of the score. */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    s64 recency_factor = 0;

    if (time_since_access < 1000000000ULL) {  /* < 1 second */
        recency_factor = 50;  /* 50% boost for very recent accesses */
    } else if (time_since_access < 10000000000ULL) {  /* 1-10 seconds */
        /* Linear decay: 50% at 1s, 0% at 10s */
        s64 elapsed_seconds = (time_since_access - 1000000000ULL) / 1000000000ULL;
        recency_factor = 50 - (elapsed_seconds * 50 / 9);  /* decay over 9 seconds */
        if (recency_factor < 0) recency_factor = 0;
    }
    /* else: time_since_access >= 10s, recency_factor remains 0 */

    s64 base_score = ((s64)meta->access_count * (100 + recency_factor)) / 100;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    }

    /* Packet loss detected: apply conservative protection boost */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_ewma > 3000) {  /* Loss + RTT >3ms */
        network_factor = (network_factor * 115) / 100;  /* 15% additional boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Per-folio heat via interval EWMA: distinguish hot vs cold folios.
     * Low interval_ewma means frequent accesses (hot).
     * High interval_ewma means infrequent accesses (cold). */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta) {
        int ret = bpf_cache_ext_list_add(main_list, folio);
        if (ret != 0)
            bpf_cache_ext_list_move(main_list, folio, false);
        struct vulcan_folio_metadata new_meta =
            vulcan_folio_init(bpf_ktime_get_ns());
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    vulcan_folio_on_access(meta, bpf_ktime_get_ns(), &folio_cfg);

    /* Collect interval statistics during first 100 eviction cycles */
    u32 stats_key = 0;
    struct {
        u64 interval_min;
        u64 interval_max;
        u64 interval_sum;
        u32 interval_count;
        u32 eviction_cycles;
    } *stats = bpf_map_lookup_elem(&interval_stats_map, &stats_key);

    if (stats && stats->eviction_cycles < 100) {
        s64 interval_val = meta->interval_ewma.value;
        if (interval_val > 0) {
            if (stats->interval_count == 0) {
                stats->interval_min = interval_val;
                stats->interval_max = interval_val;
            } else {
                if ((u64)interval_val < stats->interval_min)
                    stats->interval_min = interval_val;
                if ((u64)interval_val > stats->interval_max)
                    stats->interval_max = interval_val;
            }
            stats->interval_sum += interval_val;
            stats->interval_count += 1;
        }
    }

    bump_counter(4);
}
