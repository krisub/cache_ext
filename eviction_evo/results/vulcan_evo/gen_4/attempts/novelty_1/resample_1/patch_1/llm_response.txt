<DESCRIPTION>
The current program performs reasonably well but has room for improvement, especially on scan-heavy workloads (ycsb_c_f). The main issues are:

1. **Static bonuses don't adapt well to changing network conditions**: The fixed RTT thresholds (50ms, 20ms) and bonus amounts don't scale with actual congestion levels.

2. **Access count doesn't decay**: Old access counts from past scans persist, causing stale pages to be over-protected.

3. **Missing congestion pressure signal**: PACKETS_OUT alone isn't tracked - the ratio of packets_out/cwnd is the key indicator of network buffer pressure.

4. **Recency bonus is too generous**: Pages accessed within 30 seconds get large bonuses, which helps scans accumulate protection.

Proposed improvements:

1. **Add PACKETS_OUT listener** with RollingWindow to track in-flight packet trends
2. **Implement time-based access_count decay**: Reduce effective access count based on page age to prevent stale pages from being over-protected
3. **Use packets_out/cwnd ratio** for adaptive congestion detection instead of static RTT thresholds
4. **Reduce recency bonus magnitude** and make it decay more quickly
5. **Tighten hot folio thresholds** - current 100ms/1s thresholds are too generous for sequential scans

This approach should better distinguish between:
- Truly hot working set pages (frequent re-access, low interval)
- Scan pollution (one-time sequential access, no re-access)
- Stale pages (high old access count, but cold now)
</DESCRIPTION>

static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    // Base score: access count (higher = keep)
    s64 score = (s64)meta->access_count;

    // Network-aware protection: when RTT is high, cache misses are expensive
    // RTT is in usec; typical good RTT < 10ms (10000us), bad > 50ms (50000us)
    s64 rtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    if (rtt_ewma > 50000) {
        // High RTT: add bonus to protect frequently accessed pages
        score += meta->access_count / 2;
    } else if (rtt_ewma > 20000) {
        // Moderate RTT: small bonus
        score += meta->access_count / 4;
    }

    // Loss detection: if we've seen packet loss, be conservative
    s64 lost_max = vulcan_get_max(GF_LOST);
    if (lost_max > 0) {
        // Network congestion: protect working set
        score += meta->access_count / 3;
    }

    // Per-folio heat: lower interval EWMA = hotter (more frequent accesses)
    // Interval is in nanoseconds; < 100ms (100000000ns) = hot
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 100000000) {
        // Hot folio: frequently re-accessed, add bonus
        score += 5000;
    } else if (interval_ewma > 0 && interval_ewma < 1000000000) {
        // Warm folio: < 1 second interval
        score += 2000;
    }

    // Recency: protect recently accessed folios (likely in working set)
    u64 now = bpf_ktime_get_ns();
    s64 age = (s64)(now - meta->last_access_ts);
    if (age < 5000000000) {  // < 5 seconds
        score += 3000;
    } else if (age < 30000000000) {  // < 30 seconds
        score += 1000;
    }

    // LevelDB: protect index block (last page of file)
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    u64 now = bpf_ktime_get_ns();
    s64 age_ns = (s64)(now - meta->last_access_ts);

    // Time-based access count decay: reduce effective access count for stale pages
    // Decay factor: reduce by 50% every 10 seconds
    s64 effective_access = (s64)meta->access_count;
    if (age_ns > 10000000000) {  // > 10 seconds
        effective_access = effective_access * 3 / 4;
    }
    if (age_ns > 30000000000) {  // > 30 seconds
        effective_access = effective_access / 2;
    }
    if (age_ns > 60000000000) {  // > 60 seconds
        effective_access = effective_access / 4;
    }

    s64 score = effective_access;

    // Adaptive network-based protection using congestion pressure
    s64 packets_out = vulcan_get_ewma(GF_PACKETS_OUT);
    s64 cwnd = vulcan_get_ewma(GF_SND_CWND);
    s64 rtt_ewma = vulcan_get_ewma(GF_SRTT_US);

    // Congestion pressure: packets_out / cwnd ratio
    // High ratio = network buffers full, cache misses are expensive
    if (cwnd > 0) {
        s64 pressure_pct = (packets_out * 100) / cwnd;
        if (pressure_pct > 80) {
            // High congestion: strongly protect working set
            score += effective_access;
        } else if (pressure_pct > 60) {
            // Moderate congestion
            score += effective_access / 2;
        }
    }

    // RTT scaling: only apply if significantly elevated
    if (rtt_ewma > 40000) {  // > 40ms
        score += effective_access / 3;
    }

    // Loss detection: conservative protection
    s64 lost_max = vulcan_get_max(GF_LOST);
    if (lost_max > 0) {
        score += effective_access / 4;
    }

    // Per-folio heat: tighter thresholds to avoid protecting scan victims
    // Re-access within 50ms = truly hot working set
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 50000000) {  // < 50ms
        score += 8000;
    } else if (interval_ewma > 0 && interval_ewma < 200000000) {  // < 200ms
        score += 3000;
    }

    // Reduced recency bonus: only very recently accessed pages
    if (age_ns < 2000000000) {  // < 2 seconds
        score += 2000;
    } else if (age_ns < 10000000000) {  // < 10 seconds
        score += 500;
    }

    // LevelDB: protect index block (last page of file)
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
