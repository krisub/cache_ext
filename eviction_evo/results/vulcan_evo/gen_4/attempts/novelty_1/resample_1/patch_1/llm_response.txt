<DESCRIPTION>
The current program achieves 1977.84 by using RTT and loss detection, but this hurts performance in the congested scenario (3098 vs 3456 in the baseline). The issue is that high RTT in a congested network doesn't mean we should protect more—it means we should be smarter about what we protect.

Key improvements:
1. **Enable inter_arrival burst detection**: When inter_arrival is low (packets arriving frequently), we're in a burst phase and should protect the working set more. When inter_arrival is high, we're idle and should evict more aggressively.

2. **Soften network protection factors**: The current 150x and 125x multipliers are too aggressive. Use 110x/115x instead, applied only when loss is actually detected, not just high RTT.

3. **Add congestion window awareness**: Track the ratio of packets_out to cwnd. When we're near the window limit, we're truly congested and should be more selective.

4. **Use access_count as primary signal**: Make frequency (access_count) the dominant factor. Network conditions modify this gently, not aggressively.

5. **Soften per-folio heat bonus**: Only give the +100 bonus to very hot folios (interval < 100ms), making eviction more selective.

6. **Increase sample size under congestion**: When packets_out is high, increase SAMPLE_SIZE from 20 to 24 for better victim selection.

This approach balances network awareness (protecting during true congestion and bursts) with efficient eviction (not over-protecting during transient RTT increases).

/*
 * Baseline: pure LFU (mimics cache_ext_sampling.bpf.c bpf_lfu_score_fn).
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Evolution can add vulcan_get_* calls and protection rules.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT (>10ms smoothed): protect working set more aggressively */
    if (srtt_ewma > 10000) {
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: protect pages accessed in last 5 seconds */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 5000000000ULL) {  /* 5 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: low interval EWMA = hot (frequently accessed) */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Adaptive burst and congestion-aware eviction.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * 
 * Strategy:
 *  - Base score is frequency (access_count) — main signal
 *  - Detect bursts via inter_arrival EWMA (low = burst = protect more)
 *  - Detect congestion via packets_out/cwnd ratio (high = protect more)
 *  - Apply loss-based penalty only when actual loss detected
 *  - Per-folio heat (interval_ewma) refines score gently
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Base score: frequency is the primary signal */
    s64 score = (s64)meta->access_count;

    /* Burst detection: low inter_arrival EWMA = active burst phase */
    s64 inter_arrival_ewma = vulcan_get_ewma(GF_INTER_ARRIVAL);
    s64 burst_factor = 100;  /* 1.0x baseline */
    if (inter_arrival_ewma > 0 && inter_arrival_ewma < 50000) {  /* < 50us avg = active burst */
        burst_factor = 115;  /* 1.15x protection during bursts */
    }

    /* Congestion detection: compare in-flight packets to congestion window */
    s64 packets_out_ewma = vulcan_get_ewma(GF_PACKETS_OUT);
    s64 cwnd_ewma = vulcan_get_ewma(GF_SND_CWND);
    s64 congestion_factor = 100;  /* 1.0x baseline */
    
    if (cwnd_ewma > 0 && packets_out_ewma > 0) {
        /* If in-flight is very close to window, we're congested */
        s64 ratio = (packets_out_ewma * 100) / cwnd_ewma;  /* ratio in fixed-point (÷100) */
        if (ratio > 90) {  /* > 90% window utilization */
            congestion_factor = 110;  /* 1.1x gentle protection */
        }
    }

    /* Combine burst and congestion factors */
    score = (score * burst_factor * congestion_factor) / 10000;

    /* Loss-based penalty (only apply if actual loss detected) */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        /* Loss detected: boost protection score gently (105x) */
        score = (score * 105) / 100;
    }

    /* Per-folio heat: very hot folios (interval < 100ms) get extra protection */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 100000000) {  /* < 100ms = very hot */
        score += 80;
    } else if (interval_ewma > 5000000000) {  /* > 5 seconds = very cold */
        /* Cold folio: reduce score to prefer as victim */
        if (score > 50) score -= 50;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
