// BPF page cache eviction policy with CPU/scheduler awareness.
// Evolved: Dual-list priority segregation with adaptive aging and size-aware protection.
//
// Fixed infrastructure (outside EVOLVE-BLOCK):
//   - Includes, license, map definitions
//   - Scheduler monitoring kprobe (vfs_read) that captures CPU/scheduler metrics
//   - struct_ops registration at the end
//
// What is evolved in EVOLVE-BLOCK:
//   - Dual-list strategy (high_prio_list, low_prio_list)
//   - Adaptive time windows (RECENT_ACCESS_NS, WORKING_SET_NS)
//   - Size-aware protection for small files
//   - Priority-based eviction ordering
//   - Per-folio nice value caching for migration detection

#include "vmlinux.h"
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>
#include <bpf/bpf_core_read.h>
#include "cache_ext_lib.bpf.h"
#include "dir_watcher.bpf.h"

char _license[] SEC("license") = "GPL";

// -- Per-folio metadata -------------------------------------------------------
struct sched_folio_metadata {
    u64 last_access_ts;
    u32 access_count;
    s32 accessor_nice;
    u32 accessor_pid;
};

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __type(key, u64);
    __type(value, struct sched_folio_metadata);
    __uint(max_entries, 1000000);
} folio_metadata_map SEC(".maps");

// -- Debug counters -----------------------------------------------------------
struct {
    __uint(type, BPF_MAP_TYPE_ARRAY);
    __uint(max_entries, 16);
    __type(key, u32);
    __type(value, u64);
} debug_counters SEC(".maps");

// -- Scheduler state (updated by kprobe below) --------------------------------
u64 sched_last_read_ts = 0;
u64 sched_prev_read_ts = 0;
u64 sched_read_count = 0;

s32 sched_nice = 0;
s32 sched_prio = 0;
s32 sched_static_prio = 120;

u64 sched_vruntime = 0;
u64 sched_sum_exec_runtime = 0;
u64 sched_nr_migrations = 0;

u32 sched_nr_running = 0;
u64 sched_cfs_load = 0;

u32 sched_pid = 0;
u32 sched_tgid = 0;
u32 sched_on_cpu = 0;

// -- Helpers (fixed) ----------------------------------------------------------
static __always_inline void bump_counter(u32 idx) {
    u64 *val = bpf_map_lookup_elem(&debug_counters, &idx);
    if (val) __sync_fetch_and_add(val, 1);
}

static __always_inline bool is_folio_relevant(struct folio *folio) {
    if (!folio || !folio->mapping || !folio->mapping->host)
        return false;
    return inode_in_watchlist(folio->mapping->host->i_ino);
}

// EVOLVE-BLOCK-START
// =============================================================================
// EVICTION POLICY — Dual-list priority segregation with adaptive aging
// =============================================================================

// -- List handles + tunable constants (evolved) --------------------------------
static u64 high_prio_list = 0;  // for nice <= HIGH_PRIO_NICE_MAX
static u64 low_prio_list = 0;   // for nice > HIGH_PRIO_NICE_MAX

// Evolved time windows: shorter for faster adaptation to priority changes
#define RECENT_ACCESS_NS         30000000ULL   // 30 ms (was 100ms) — faster responsiveness
#define WORKING_SET_NS          300000000ULL   // 300 ms (was 1s) — adapts to priority flips
#define HOT_ACCESS_THRESHOLD      2             // (was 3) — easier to be "hot"
#define VERY_HOT_ACCESS_THRESHOLD 6             // Inviolable unless very old
#define HIGH_PRIO_NICE_MAX       -5             // (was 0) — only truly high-priority
#define SMALL_FILE_SIZE_BYTES   65536ULL        // 64 KB threshold for extra protection
#define CPU_BUSY_THRESHOLD       4              // nr_running > 4 = busy

// -- Helper: Is this folio small and should get extra protection?
static __always_inline bool should_protect_small_file(struct folio *folio)
{
    u64 folio_size = folio_nr_pages(folio) * 4096;
    return folio_size <= SMALL_FILE_SIZE_BYTES;
}

// -- Helper: Determine which list this folio belongs to
static __always_inline bool is_high_prio_folio(s32 accessor_nice)
{
    return accessor_nice <= HIGH_PRIO_NICE_MAX;
}

// -- Eviction callback for low-priority list (evict more aggressively) -------
static int bpf_evict_low_prio_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Never evict dirty/writeback pages
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age = now - meta->last_access_ts;

    // Very hot folios (6+ accesses) get more protection
    if (meta->access_count >= VERY_HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Hot folios (2+ accesses) get standard working-set protection
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Recently accessed pages: more lenient window for low-prio
    if (age < WORKING_SET_NS)
        return CACHE_EXT_CONTINUE_ITER;

    // Low-priority pages aged beyond working set: evict
    return CACHE_EXT_EVICT_NODE;
}

// -- Eviction callback for high-priority list (protect aggressively) --------
static int bpf_evict_high_prio_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Never evict dirty/writeback pages
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age = now - meta->last_access_ts;

    // Very hot folios: protect aggressively
    if (meta->access_count >= VERY_HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 3)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Hot folios from high-priority processes: protect well
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Small files from high-priority: extend protection window
    bool is_small = should_protect_small_file(node->folio);
    if (is_small) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // When CPU is busy, protect high-prio working set aggressively
    if (sched_nr_running > CPU_BUSY_THRESHOLD) {
        if (age < WORKING_SET_NS)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Recently accessed: protect
    if (age < RECENT_ACCESS_NS)
        return CACHE_EXT_CONTINUE_ITER;

    // Only evict if very old
    if (age < WORKING_SET_NS)
        return CACHE_EXT_CONTINUE_ITER;

    return CACHE_EXT_EVICT_NODE;
}

// -- struct_ops: init ---------------------------------------------------------
s32 BPF_STRUCT_OPS_SLEEPABLE(evolved_init, struct mem_cgroup *memcg)
{
    high_prio_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!high_prio_list)
        return -1;

    low_prio_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!low_prio_list)
        return -1;

    return 0;
}

// -- struct_ops: evict_folios -------------------------------------------------
void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    bump_counter(0);
    
    // Evict from low-priority list first
    bpf_cache_ext_list_iterate(memcg, low_prio_list, bpf_evict_low_prio_cb, eviction_ctx);
    
    // Only if needed, evict from high-priority list
    bpf_cache_ext_list_iterate(memcg, high_prio_list, bpf_evict_high_prio_cb, eviction_ctx);
}

// -- struct_ops: folio_added --------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_added, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    // Get current process priority
    struct task_struct *task = (struct task_struct *)bpf_get_current_task();
    s32 nice = BPF_CORE_READ(task, static_prio) - 120;
    u32 pid  = bpf_get_current_pid_tgid() >> 32;

    // Route to appropriate list
    u64 target_list = is_high_prio_folio(nice) ? high_prio_list : low_prio_list;
    int ret = bpf_cache_ext_list_add_tail(target_list, folio);
    if (ret != 0) {
        bump_counter(1);
        return;
    }

    // Record metadata
    u64 key = (u64)folio;
    struct sched_folio_metadata new_meta = {
        .last_access_ts = bpf_ktime_get_ns(),
        .access_count   = 1,
        .accessor_nice  = nice,
        .accessor_pid   = pid,
    };
    bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
    bump_counter(2);
}

// -- struct_ops: folio_accessed -----------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    struct task_struct *task = (struct task_struct *)bpf_get_current_task();
    s32 nice = BPF_CORE_READ(task, static_prio) - 120;
    u32 pid  = bpf_get_current_pid_tgid() >> 32;

    u64 key = (u64)folio;
    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    bool is_high_prio = is_high_prio_folio(nice);
    u64 target_list = is_high_prio ? high_prio_list : low_prio_list;

    if (!meta) {
        // Folio not yet tracked; add it
        int ret = bpf_cache_ext_list_add(target_list, folio);
        if (ret != 0) {
            bpf_cache_ext_list_move(target_list, folio, false);
        }
        struct sched_folio_metadata new_meta = {
            .last_access_ts = bpf_ktime_get_ns(),
            .access_count   = 1,
            .accessor_nice  = nice,
            .accessor_pid   = pid,
        };
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    // Update metadata
    meta->last_access_ts = bpf_ktime_get_ns();
    meta->access_count += 1;
    meta->accessor_nice = nice;
    meta->accessor_pid  = pid;

    // Move to head (MRU) within the appropriate list
    bpf_cache_ext_list_move(target_list, folio, false);
    bump_counter(4);
}

// -- struct_ops: folio_evicted ------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_evicted, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    // Delete from whichever list it's on (try both to be safe)
    bpf_cache_ext_list_del(folio);

    u64 key = (u64)folio;
    bpf_map_delete_elem(&folio_metadata_map, &key);
    bump_counter(5);
}

// =============================================================================
// SCHEDULER MONITORING (fixed — not evolved)
// =============================================================================

SEC("kprobe/vfs_read")
int trace_vfs_read(struct pt_regs *ctx)
{
    struct file *f = (struct file *)PT_REGS_PARM1(ctx);
    if (!f) return 0;

    u64 ino = BPF_CORE_READ(f, f_inode, i_ino);
    if (!inode_in_watchlist(ino)) return 0;

    // Timing
    sched_prev_read_ts = sched_last_read_ts;
    sched_last_read_ts = bpf_ktime_get_ns();
    sched_read_count += 1;

    // Read scheduler metrics from current task
    struct task_struct *task = (struct task_struct *)bpf_get_current_task();

    // Process priority / niceness
    sched_static_prio = BPF_CORE_READ(task, static_prio);
    sched_prio        = BPF_CORE_READ(task, prio);
    sched_nice        = sched_static_prio - 120;

    // CPU execution metrics from sched_entity
    sched_vruntime         = BPF_CORE_READ(task, se.vruntime);
    sched_sum_exec_runtime = BPF_CORE_READ(task, se.sum_exec_runtime);
    sched_nr_migrations    = BPF_CORE_READ(task, se.nr_migrations);

    // Run queue state
    struct cfs_rq *cfs = BPF_CORE_READ(task, se.cfs_rq);
    if (cfs) {
        sched_nr_running = BPF_CORE_READ(cfs, nr_running);
        sched_cfs_load   = BPF_CORE_READ(cfs, load.weight);
    }

    // Process identification
    u64 pid_tgid = bpf_get_current_pid_tgid();
    sched_pid    = (u32)pid_tgid;
    sched_tgid   = (u32)(pid_tgid >> 32);
    sched_on_cpu = BPF_CORE_READ(task, on_cpu);

    return 0;
}

// -- struct_ops registration --------------------------------------------------
SEC(".struct_ops.link")
struct cache_ext_ops evolved_ops = {
    .init           = (void *)evolved_init,
    .evict_folios   = (void *)evolved_evict_folios,
    .folio_added    = (void *)evolved_folio_added,
    .folio_evicted  = (void *)evolved_folio_evicted,
    .folio_accessed = (void *)evolved_folio_accessed,
    .admit_folio    = (void *)0,
};
