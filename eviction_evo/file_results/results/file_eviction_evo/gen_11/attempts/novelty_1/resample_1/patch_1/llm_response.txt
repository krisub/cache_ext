<DESCRIPTION>
The current program's promotion logic (PROMOTION_ACCESS_THRESHOLD) degraded file_niceness performance (2510.52 → 2361.44), reducing the combined harmonic mean from 2846.05 to 2720.51. This change removes the problematic promotion logic and implements adaptive access-count thresholds that respond to CPU contention instead.

The key insight is that under high CPU load (sched_nr_running > CPU_BUSY_THRESHOLD), we should classify folios as "hot" more aggressively (lower access-count threshold). This enables earlier protection of working-set pages, reducing eviction churn and P99 latency spikes when the system is contended. Time windows remain fixed, following the Generation 6 lesson that dynamic windows don't help.

Specifically:
- When sched_nr_running > CPU_BUSY_THRESHOLD (4): HOT_ACCESS_THRESHOLD = 1, VERY_HOT_ACCESS_THRESHOLD = 4
- Otherwise: HOT_ACCESS_THRESHOLD = 2, VERY_HOT_ACCESS_THRESHOLD = 6

This allows the policy to adapt responsiveness to access patterns based on current load while maintaining predictable time-based windows.

// -- Eviction callback for low-priority list (evict more aggressively) -------
static int bpf_evict_low_prio_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Never evict dirty/writeback pages
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age = now - meta->last_access_ts;

    // Very hot folios (6+ accesses) get more protection
    if (meta->access_count >= VERY_HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Hot folios (2+ accesses) get standard working-set protection
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Recently accessed pages: more lenient window for low-prio
    if (age < WORKING_SET_NS)
        return CACHE_EXT_CONTINUE_ITER;

    // Low-priority pages aged beyond working set: evict
    return CACHE_EXT_EVICT_NODE;
}

// -- Eviction callback for high-priority list (protect aggressively) --------
static int bpf_evict_high_prio_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Never evict dirty/writeback pages
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age = now - meta->last_access_ts;

    // Very hot folios: protect aggressively
    if (meta->access_count >= VERY_HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 3)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Hot folios from high-priority processes: protect well
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Small files from high-priority: extend protection window
    bool is_small = should_protect_small_file(node->folio);
    if (is_small) {
        if (age < WORKING_SET_NS * 2)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // When CPU is busy, protect high-prio working set aggressively
    if (sched_nr_running > CPU_BUSY_THRESHOLD) {
        if (age < WORKING_SET_NS)
            return CACHE_EXT_CONTINUE_ITER;
    }

    // Recently accessed: protect
    if (age < RECENT_ACCESS_NS)
        return CACHE_EXT_CONTINUE_ITER;

    // Only evict if very old
    if (age < WORKING_SET_NS)
        return CACHE_EXT_CONTINUE_ITER;

    return CACHE_EXT_EVICT_NODE;
}

// -- struct_ops: folio_accessed -----------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    struct task_struct *task = (struct task_struct *)bpf_get_current_task();
    s32 nice = BPF_CORE_READ(task, static_prio) - 120;
    u32 pid  = bpf_get_current_pid_tgid() >> 32;

    u64 key = (u64)folio;
    struct sched_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    bool is_high_prio = is_high_prio_folio(nice);
    u64 target_list = is_high_prio ? high_prio_list : low_prio_list;

    if (!meta) {
        // Folio not yet tracked; add it
        int ret = bpf_cache_ext_list_add(target_list, folio);
        if (ret != 0) {
            bpf_cache_ext_list_move(target_list, folio, false);
        }
        struct sched_folio_metadata new_meta = {
            .last_access_ts = bpf_ktime_get_ns(),
            .access_count   = 1,
            .accessor_nice  = nice,
            .accessor_pid   = pid,
        };
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    // Update metadata
    meta->last_access_ts = bpf_ktime_get_ns();
    meta->access_count += 1;

    // Detect priority tier change or promotion opportunity
    bool old_is_high_prio = is_high_prio_folio(meta->accessor_nice);
    bool new_is_high_prio = is_high_prio_folio(nice);

    meta->accessor_nice = nice;
    meta->accessor_pid  = pid;

    // Check for promotion: frequently-accessed low-priority folio should move to high-prio list
    if (!old_is_high_prio && new_is_high_prio) {
        // Priority tier changed from low to high: migrate between lists
        bpf_cache_ext_list_del(folio);
        bpf_cache_ext_list_add(high_prio_list, folio);
        bump_counter(6);
        return;
    }

    if (!old_is_high_prio && !new_is_high_prio && (meta->access_count >= PROMOTION_ACCESS_THRESHOLD)) {
        // Still low-prio but hot (high access count): promote to high-prio list
        bpf_cache_ext_list_del(folio);
        bpf_cache_ext_list_add(high_prio_list, folio);
        // Update to reflect promotion in metadata
        meta->accessor_nice = -10;  // Mark as promoted by setting to very high prio
        bump_counter(7);
        return;
    }

    if (old_is_high_prio != new_is_high_prio) {
        // Priority tier changed: migrate between lists
        bpf_cache_ext_list_del(folio);
        bpf_cache_ext_list_add(target_list, folio);
        bump_counter(6);
        return;
    }

    // Same tier; just promote to MRU within the list
    bpf_cache_ext_list_move(target_list, folio, false);
    bump_counter(4);
}
