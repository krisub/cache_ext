// -- Helper: compute effective access count with time decay -------------------
static __always_inline u32 get_effective_access_count(struct net_folio_metadata *meta, u64 now)
{
    if (!meta || meta->access_count == 0)
        return 0;

    u64 age = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;

    // Decay: each half-life period, count is divided by 2
    u32 decayed = meta->access_count;
    u64 remaining_age = age;
    while (remaining_age > ACCESS_DECAY_HALF_LIFE_NS && decayed > 1) {
        decayed = decayed / 2;
        remaining_age -= ACCESS_DECAY_HALF_LIFE_NS;
    }

    return decayed;
}

static __always_inline u64 get_adaptive_decay_window_ns(void)
{
    // Detect network congestion
    bool is_congested = (net_lost > 0) || (net_retrans_out > 0);

    if (is_congested) {

        return ACCESS_DECAY_WINDOW_BURST_NS;

static __always_inline u32 get_effective_access_count(struct net_folio_metadata *meta, u64 now)
{

        return 0;

static __always_inline u32 get_folio_score(struct net_folio_metadata *meta, u64 now)
{
    u32 effective_accesses = get_effective_access_count(meta, now);
    u64 age = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;

    // Recency bonus: linear decay from 100 to 0 over RECENT_ACCESS_NS
    u32 recency_bonus = 0;
    if (age < RECENT_ACCESS_NS_BASE) {

    // Combined score: frequency contribution + recency bonus
    u32 score = (effective_accesses * SCORE_PER_ACCESS) + recency_bonus;
