<DESCRIPTION>
The current program uses exponential time-decay for loss protection and multiplies access_count by a recency_factor, which adds unnecessary complexity. The best-performing program (2184.76) uses a simpler additive recency boost and MinMax-based loss detection.

The key improvement is to refine the cold folio penalty with access-count-scaled asymmetry. Currently, all cold folios (interval_ewma > 10s) receive a fixed -100 penalty. This over-penalizes long-lived but infrequently-accessed metadata structures like LevelDB index blocks that have high access_count but long inter-access intervals.

The proposed change:
- Keep the simple additive recency boost (+50 for <1s, linearly decaying to 0 at 10s)
- For cold folios (interval_ewma > 10s), scale the penalty based on access history:
  - access_count < 5: apply -100 (genuinely cold, one-shot accesses)
  - access_count < 20: apply -50 (moderately used over time)
  - access_count >= 20: apply -25 (long-lived, proven utility)

This preserves folios with demonstrated utility while still preferring cold pages as victims, reducing variance in dual_small_vs_large (std: 217.25 in best program) by avoiding premature eviction of valuable long-lived pages.
</DESCRIPTION>

/*
 * Network-aware LFU with loss-triggered protection using exponential time-decay.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Loss boost decays over ~300ms to prevent sustained over-protection from stale signals.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Recency-weighted access normalization: multiply access_count by time-decay factor
     * to prioritize recently-active folios. This scales protection with actual hotness
     * rather than applying a flat bonus that treats all recent folios equally.
     */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    s64 recency_factor = 100;  /* baseline: 1.0x multiplier */

    if (time_since_access < 1000000000ULL) {  /* < 1 second: very fresh */
        recency_factor = 150;  /* 1.5x multiplier (50% bonus) */
    } else if (time_since_access < 10000000000ULL) {  /* 1-10 seconds: decay linearly */
        /* Linear interpolation from 150 (at 1s) down to 100 (at 10s) */
        s64 elapsed_s = (time_since_access - 1000000000ULL) / 1000000000ULL;  /* 0-9 range */
        recency_factor = 150 - (elapsed_s * 5);  /* Decreases by 5 per second: 150→145→...→105→100 */
        if (recency_factor < 100) recency_factor = 100;  /* Floor at 100 */
    }
    /* else: time_since_access >= 10s, recency_factor stays 100 (no bonus) */

    s64 base_score = ((s64)meta->access_count * recency_factor) / 100;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected with time-decay protection:
     * Apply 15% boost only when loss is fresh AND RTT is elevated.
     * Boost decays exponentially: after 100ms it's ~37% of peak, after 300ms negligible.
     */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_ewma > 3000) {  /* Loss + RTT > 3ms */
        u64 time_since_loss = now - last_loss_ts;

        /* Exponential decay: decay_factor = 100 * exp(-(time_since_loss / 100ms))
         * Approximated: decay_factor = 100 / (1 + time_since_loss / 100ms)
         * This gives ~37% at 100ms, ~9% at 300ms, negligible after 500ms
         */
        s64 decay_factor = 100;
        if (time_since_loss > 0 && time_since_loss < 500000000) {  /* Within 500ms window */
            s64 elapsed_ms = time_since_loss / 1000000;  /* Convert ns to ms */
            decay_factor = (100 * 100) / (100 + elapsed_ms);  /* Approximately exponential */
            if (decay_factor < 1) decay_factor = 1;  /* Floor at 1 */
        } else if (time_since_loss >= 500000000) {
            decay_factor = 1;  /* Negligible boost after 500ms */
        }

        network_factor = (network_factor * (100 + (15 * decay_factor / 100))) / 100;
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Per-folio heat: use interval EWMA to distinguish hot vs cold folios.
     * Low interval_ewma = frequent accesses (hot).
     * High interval_ewma = infrequent accesses (cold).
     */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Network-aware LFU with sophisticated adaptive protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 *
 * Combines RTT sensitivity, per-folio heat tracking (via interval_ewma),
 * and recency with access-count-scaled cold folio penalty to preserve
 * long-lived metadata structures while still preferring cold pages as victims.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    /* Recency-weighted base score: prioritize recently-active folios over cold ones
     * with similar historical access counts. Linear decay from +50% boost (< 1 second)
     * to 0% boost (10+ seconds). This captures working-set dynamics by making temporal
     * locality a first-class component of the score. */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    s64 recency_factor = 0;

    if (time_since_access < 1000000000ULL) {  /* < 1 second */
        recency_factor = 50;  /* 50% boost for very recent accesses */
    } else if (time_since_access < 10000000000ULL) {  /* 1-10 seconds */
        /* Linear decay: 50% at 1s, 0% at 10s */
        s64 elapsed_seconds = (time_since_access - 1000000000ULL) / 1000000000ULL;
        recency_factor = 50 - (elapsed_seconds * 50 / 9);  /* decay over 9 seconds */
        if (recency_factor < 0) recency_factor = 0;
    }
    /* else: time_since_access >= 10s, recency_factor remains 0 */

    s64 base_score = ((s64)meta->access_count * (100 + recency_factor)) / 100;

    /* Network-aware protection: boost when RTT is high (cache misses are expensive) */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT (>5ms smoothed): protect working set more aggressively */
    if (srtt_ewma > 5000) {
        network_factor = 125;  /* 1.25x protection */
    }
    if (srtt_ewma > 10000) {
        network_factor = 150;  /* 1.5x protection for >10ms RTT */
    }

    /* Packet loss detected: apply conservative protection boost.
     * Only boost when loss is detected AND RTT is elevated to avoid
     * stale loss signals causing sustained over-protection. */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_ewma > 3000) {  /* Loss + RTT >3ms */
        network_factor = (network_factor * 115) / 100;  /* 15% additional boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Per-folio heat via interval EWMA: distinguish hot vs cold folios.
     * Low interval_ewma means frequent accesses (hot).
     * High interval_ewma means infrequent accesses (cold). */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: scale penalty based on access_count to preserve
         * long-lived moderately-accessed folios (e.g., LevelDB index structures).
         * - access_count < 5: genuinely cold, apply -100
         * - access_count < 20: moderately old but used, apply -50
         * - access_count >= 20: long-lived and frequently accessed, apply -25
         * This avoids over-penalizing pages with demonstrated utility over time. */
        s64 cold_penalty = 100;
        if (meta->access_count >= 20) {
            cold_penalty = 25;
        } else if (meta->access_count >= 5) {
            cold_penalty = 50;
        }
        if (score > cold_penalty) score -= cold_penalty;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
