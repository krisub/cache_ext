<DESCRIPTION>
The current program uses adaptive interval thresholds collected during early eviction cycles, but this adds complexity and may have race conditions in the statistics updates. Looking at the best prior performer (1977.84 score), it used simpler fixed thresholds and pure network-aware protection.

The recommendation to integrate PACKETS_OUT congestion signal with dynamic boost scaling based on cwnd utilization ratio is well-motivated: when packets_out approaches snd_cwnd, the congestion window is saturated, indicating network backpressure. In such situations, protecting the working set more aggressively reduces cache misses during periods when each miss is especially expensive.

This edit:
1. **Enables GF_SND_CWND EWMA listener** to track congestion window smoothly
2. **Simplifies scoring** by removing the problematic adaptive threshold collection (which adds BPF state complexity and potential races) and reverting to fixed, proven thresholds
3. **Adds window saturation boost** in bpf_score_fn: computes packets_out_ewma / snd_cwnd_ewma ratio and scales network_factor by 1.05x (if ratio > 0.7) or 1.15x (if ratio > 0.9)
4. **Maintains per-folio interval tracking** (which was beneficial in prior programs) to detect hot vs cold folios

The congestion window saturation ratio is a direct signal of how close we are to TCP send limits—much more precise than static thresholds. Combined with RTT-based and loss-based protection, this provides adaptive, multi-signal eviction decisions that should improve performance on both small-vs-large (fast readers need protection when network is busy) and congested (both clients compete for limited window) workloads.

/*
 * Simplified LFU with workload-adaptive interval thresholds and network-aware protection.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Thresholds adapt based on observed per-folio interval_ewma distribution.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: use interval EWMA with adaptive thresholds based on workload.
     * Compute dynamic hot/cold thresholds from observed interval_ewma statistics.
     * Default fallback if no statistics yet: hot if < 1s, cold if > 10s.
     */
    s64 interval_ewma = meta->interval_ewma.value;

    u32 stats_key = 0;
    struct {
        u64 interval_min;
        u64 interval_max;
        u64 interval_sum;
        u32 interval_count;
        u32 eviction_cycles;
    } *stats = bpf_map_lookup_elem(&interval_stats_map, &stats_key);

    s64 hot_threshold = 1000000000;      /* Default: < 1 second */
    s64 cold_threshold = 10000000000;    /* Default: > 10 seconds */

    if (stats && stats->interval_count > 10) {
        /* Compute adaptive thresholds from observed distribution */
        s64 mean_interval = stats->interval_sum / stats->interval_count;
        /* Hot threshold: 0.5x median/mean (more frequent than average) */
        hot_threshold = (mean_interval / 2);
        if (hot_threshold < 500000000) hot_threshold = 500000000;  /* Floor at 0.5s */
        /* Cold threshold: 5x median/mean (less frequent than average) */
        cold_threshold = (mean_interval * 5);
        if (cold_threshold > 15000000000ULL) cold_threshold = 15000000000ULL;  /* Cap at 15s */
    }

    if (interval_ewma > 0 && interval_ewma < hot_threshold) {
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > cold_threshold) {
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * LFU with congestion window saturation boost and per-folio heat tracking.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Uses multi-signal network awareness: RTT, packet loss, and window saturation.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Multi-signal network-aware protection */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 packets_out_ewma = vulcan_get_ewma(GF_PACKETS_OUT);
    s64 cwnd_ewma = vulcan_get_ewma(GF_SND_CWND);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Signal 1: Elevated RTT: protect working set more aggressively */
    if (srtt_ewma > 10000) {  /* >10ms smoothed RTT */
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {  /* >5ms smoothed RTT */
        network_factor = 125;  /* 1.25x protection */
    }

    /* Signal 2: Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Signal 3: Congestion window saturation via packets_out / cwnd ratio.
     * When ratio approaches 1.0, window is saturated and we should protect
     * the working set more aggressively to avoid eviction during backpressure.
     * Use safe integer arithmetic to compute ratio.
     */
    if (cwnd_ewma > 0 && packets_out_ewma > 0) {
        /* Compute ratio as: (packets_out * 1000) / cwnd to avoid precision loss */
        s64 saturation_ratio = (packets_out_ewma * 1000) / cwnd_ewma;
        if (saturation_ratio > 900) {
            /* Ratio > 0.9: window nearly full, apply 1.15x boost */
            network_factor = (network_factor * 115) / 100;
        } else if (saturation_ratio > 700) {
            /* Ratio 0.7-0.9: moderate window usage, apply 1.05x boost */
            network_factor = (network_factor * 105) / 100;
        }
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: boost pages accessed very recently (< 3 seconds) */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 3000000000ULL) {  /* 3 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: low interval EWMA = hot (frequently accessed) */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses, protect strongly */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
