<DESCRIPTION>
Implement adaptive RTT baseline tracking with dynamic threshold scaling to improve performance across workloads. The current program applies a fixed 30% loss boost unconditionally, which may over-protect during low-RTT workloads like dual_small_vs_large and under-protect during congested scenarios.

The recommendation suggests maintaining a separate long-term EWMA baseline of SRTT (with low smoothing alpha around 50) to establish a "normal" RTT profile for the workload. Then dynamically scale protection factors based on how much current RTT deviates from this baseline:
- When current RTT > baseline × 1.2: apply 1.25x protection multiplier
- When current RTT > baseline × 1.5: apply 1.5x protection multiplier  
- Loss boost only applied when both loss is detected AND current RTT > baseline × 1.1

This avoids over-protection during naturally low-RTT workloads and focuses protection efforts during actual degradation.

Changes:
1. Add a map to track long-term RTT baseline EWMA
2. Update kprobe to compute baseline EWMA alongside listeners
3. Modify bpf_score_fn to dynamically scale thresholds based on RTT deviation from baseline
4. Reduce unconditional loss boost to conditional baseline-relative boost
</DESCRIPTION>

/*
 * Baseline: pure LFU (mimics cache_ext_sampling.bpf.c bpf_lfu_score_fn).
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 * Evolution can add vulcan_get_* calls and protection rules.
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Network-aware protection: when RTT is high, cache misses are expensive */
    s64 srtt_ewma = vulcan_get_ewma(GF_SRTT_US);
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Elevated RTT (>10ms smoothed): protect working set more aggressively */
    if (srtt_ewma > 10000) {
        network_factor = 150;  /* 1.5x protection */
    } else if (srtt_ewma > 5000) {
        network_factor = 125;  /* 1.25x protection */
    }

    /* Packet loss detected: be more conservative */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0) {
        network_factor = (network_factor * 130) / 100;  /* additional 30% boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: protect pages accessed in last 5 seconds */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 5000000000ULL) {  /* 5 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: low interval EWMA = hot (frequently accessed) */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}

/*
 * Adaptive RTT baseline-aware LFU with dynamic threshold scaling.
 * Kernel picks LOWEST score in each batch. S64_MAX = never prefer as victim.
 *
 * Uses short-term EWMA (alpha=150) for current RTT and long-term EWMA (alpha=50)
 * baseline to detect RTT deviations. Protection scales dynamically:
 * - When current > baseline*1.2: 1.25x protection
 * - When current > baseline*1.5: 1.5x protection
 * - Loss boost only when both loss detected AND current > baseline*1.1
 */
static s64 bpf_score_fn(struct cache_ext_list_node *node)
{
    struct folio *folio = node->folio;
    if (!folio)
        return S64_MAX;

    if (folio_test_dirty(folio) || folio_test_writeback(folio))
        return S64_MAX;
    if (!folio_test_uptodate(folio) || !folio_test_lru(folio))
        return S64_MAX;

    u64 key = (u64)folio;
    struct vulcan_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);
    if (!meta)
        return S64_MAX;

    s64 base_score = (s64)meta->access_count;

    /* Adaptive RTT protection based on baseline deviation */
    s64 srtt_current = vulcan_get_ewma(GF_SRTT_US);
    
    /* Retrieve long-term baseline RTT */
    u32 idx = 0;
    u64 *baseline_ptr = bpf_map_lookup_elem(&srtt_baseline_map, &idx);
    s64 srtt_baseline = baseline_ptr ? (s64)(*baseline_ptr) : srtt_current;
    
    s64 network_factor = 100;  /* baseline: 1.0x in fixed-point (÷100) */

    /* Apply dynamic thresholds based on deviation from baseline.
     * Only scale up protection when current RTT exceeds baseline,
     * avoiding over-protection during naturally low-RTT workloads. */
    if (srtt_baseline > 0) {
        /* Check if current RTT has elevated significantly (>1.5x baseline) */
        if (srtt_current > (srtt_baseline * 150) / 100) {
            network_factor = 150;  /* 1.5x protection during major degradation */
        } 
        /* Check if current RTT moderately elevated (>1.2x baseline) */
        else if (srtt_current > (srtt_baseline * 120) / 100) {
            network_factor = 125;  /* 1.25x protection during moderate elevation */
        }
    } else {
        /* Fallback to absolute thresholds if baseline not yet established */
        if (srtt_current > 10000) {
            network_factor = 150;  /* 1.5x protection for >10ms RTT */
        } else if (srtt_current > 5000) {
            network_factor = 125;  /* 1.25x protection for >5ms RTT */
        }
    }

    /* Packet loss with adaptive boost: only boost when loss + elevated RTT */
    s64 max_lost = vulcan_get_max(GF_LOST);
    if (max_lost > 0 && srtt_baseline > 0) {
        /* Apply loss boost only when current RTT > baseline*1.1 (10% elevation) */
        if (srtt_current > (srtt_baseline * 110) / 100) {
            network_factor = (network_factor * 115) / 100;  /* 15% additional boost */
        }
    } else if (max_lost > 0 && srtt_current > 3000) {
        /* Fallback to absolute RTT threshold if baseline not available */
        network_factor = (network_factor * 115) / 100;  /* 15% additional boost */
    }

    /* Apply network-aware protection factor */
    s64 score = (base_score * network_factor) / 100;

    /* Recency bonus: protect pages accessed in last 5 seconds */
    u64 now = bpf_ktime_get_ns();
    u64 time_since_access = now - meta->last_access_ts;
    if (time_since_access < 5000000000ULL) {  /* 5 seconds in ns */
        score += 50;  /* recency boost */
    }

    /* Per-folio heat: low interval EWMA = hot (frequently accessed) */
    s64 interval_ewma = meta->interval_ewma.value;
    if (interval_ewma > 0 && interval_ewma < 1000000000) {  /* < 1 second */
        /* Hot folio: very frequent accesses */
        score += 100;
    } else if (interval_ewma > 10000000000) {  /* > 10 seconds */
        /* Cold folio: infrequent accesses, prefer as victim */
        if (score > 100) score -= 100;
    }

    /* LevelDB: protect index block (last page of file) */
    if (is_last_page_in_file(folio))
        score += 100000;

    return score;
}
