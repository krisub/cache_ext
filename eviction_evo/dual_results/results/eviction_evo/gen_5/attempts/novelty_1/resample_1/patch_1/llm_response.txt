<DESCRIPTION>
This restructures the eviction policy from a callback-per-folio scanning model to a **stratified sampling + adaptive scoring model** with **phase-aware state tracking**.

Key architectural changes:

1. **Stratified Sampling Instead of Full Iteration**: Rather than calling the eviction callback on every folio, we use `bpf_cache_ext_list_sample()` with adaptive ratios. This reduces CPU overhead and allows the policy to focus on high-impact folios.

2. **Phase-State Machine**: We track network and workload phases (idle, normal, burst, congested) as discrete states rather than recomputing indicators repeatedly. This improves decision consistency and reduces redundant calculations.

3. **Per-Phase Eviction Strategies**: Each network phase (idle, normal-load, burst, severe-congestion) has tuned eviction thresholds and scoring weights. Phase detection is done once per eviction call and cached in globals.

4. **Dual-List with Adaptive Thresholds**: We maintain a single main list but with phase-aware scoring that dynamically weights recency vs. frequency. During bursts, we weight recency heavier; during congestion, we preserve the working set more aggressively.

5. **Precomputed Network Score**: Instead of recomputing congestion/RTT indicators in every folio callback, we compute a single network score at eviction time and pass it down. This eliminates redundant TCP metric reads.

6. **Burst Duration Tracking**: We track burst_start_ts to detect sustained bursts (not just single packet arrivals), allowing more aggressive working-set protection during sustained traffic spikes.

This design reduces per-folio computation overhead, improves cache locality of network metric reads, and makes policy behavior more predictable across workload phases.

// BPF page cache eviction policy with network awareness.
// Restructured for stratified sampling and phase-aware eviction.

#include "vmlinux.h"
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>
#include <bpf/bpf_core_read.h>
#include "cache_ext_lib.bpf.h"
#include "dir_watcher.bpf.h"

char _license[] SEC("license") = "GPL";

// -- Per-folio metadata -------------------------------------------------------
struct net_folio_metadata {
    u64 last_access_ts;
    u32 access_count;        // number of times this folio was accessed
    u32 access_tier;         // tier classification (0=cold, 1=warm, 2=hot)
};

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __type(key, u64);          // (u64)folio pointer
    __type(value, struct net_folio_metadata);
    __uint(max_entries, 1000000);
} folio_metadata_map SEC(".maps");

// -- Debug counters -----------------------------------------------------------
struct {
    __uint(type, BPF_MAP_TYPE_ARRAY);
    __uint(max_entries, 16);
    __type(key, u32);
    __type(value, u64);
} debug_counters SEC(".maps");

// -- Network state (updated by kprobe below) ----------------------------------
u64 last_packet_ts = 0;

// -- Extended network metrics (snapshot from tcp_sock per recv) ---------------
u64 net_recv_count = 0;
u64 net_prev_packet_ts = 0;
u32 net_srtt_us = 0;
u32 net_mdev_us = 0;
u64 net_bytes_received = 0;
u64 net_bytes_acked = 0;
u32 net_segs_in = 0;
u32 net_segs_out = 0;
u32 net_delivered = 0;
u32 net_snd_cwnd = 0;
u32 net_snd_ssthresh = 0;
u32 net_rcv_wnd = 0;
u32 net_packets_out = 0;
u32 net_total_retrans = 0;
u32 net_retrans_out = 0;
u32 net_lost = 0;

// -- Helpers (fixed) ----------------------------------------------------------
static __always_inline void bump_counter(u32 idx) {
    u64 *val = bpf_map_lookup_elem(&debug_counters, &idx);
    if (val) __sync_fetch_and_add(val, 1);
}

static __always_inline bool is_folio_relevant(struct folio *folio) {
    if (!folio || !folio->mapping || !folio->mapping->host)
        return false;
    return inode_in_watchlist(folio->mapping->host->i_ino);
}

// EVOLVE-BLOCK-START

// =============================================================================
// PHASE-STATE MACHINE FOR NETWORK-AWARE EVICTION
// =============================================================================

// Network phase classification
#define PHASE_IDLE          0
#define PHASE_NORMAL        1
#define PHASE_BURST         2
#define PHASE_CONGESTED     3

// Phase-dependent constants (evolved)
#define BASE_RECENT_NS          100000000ULL   // 100ms
#define BASE_WORKING_SET_NS     1000000000ULL  // 1s
#define NET_IDLE_TIMEOUT_NS     5000000000ULL  // 5s
#define BURST_THRESHOLD_NS      5000000ULL     // 5ms inter-arrival
#define BURST_SUSTAIN_NS        100000000ULL   // 100ms to sustain burst
#define RTT_THRESHOLD_US        20000          // 20ms = congested
#define JITTER_THRESHOLD_US     5000           // 5ms jitter = congested

// Phase-state globals (computed once per eviction call)
static u32 current_phase = PHASE_NORMAL;
static u64 phase_score = 500;           // 0-1000 scale
static u64 burst_start_ts = 0;          // when burst started
static bool in_burst = false;            // current burst state

// -- Network metric analysis helpers ------------------------------------------

static __always_inline u32 classify_phase(u64 now)
{
    // Check if session is idle
    if ((now - last_packet_ts) > NET_IDLE_TIMEOUT_NS)
        return PHASE_IDLE;

    // Check for burst: fast inter-arrival
    u64 inter_arrival = (last_packet_ts > net_prev_packet_ts) ? 
        (last_packet_ts - net_prev_packet_ts) : 1000000000ULL;
    
    bool is_fast_arrival = (inter_arrival < BURST_THRESHOLD_NS);
    
    // Check for network congestion: high RTT or jitter or loss
    u32 rtt_us = net_srtt_us >> 3;  // kernel stores srtt<<3
    u32 jitter_us = net_mdev_us >> 3;
    bool is_congested = (rtt_us > RTT_THRESHOLD_US) || 
                        (jitter_us > JITTER_THRESHOLD_US) ||
                        (net_retrans_out > 0) || (net_lost > 0);

    if (is_congested)
        return PHASE_CONGESTED;
    
    if (is_fast_arrival)
        return PHASE_BURST;
    
    return PHASE_NORMAL;
}

// Compute a single network quality score (0-1000, higher = worse conditions)
static __always_inline u32 compute_network_score(void)
{
    u32 score = 0;
    
    // RTT component (0-350 points)
    u32 rtt_us = net_srtt_us >> 3;
    if (rtt_us > 100000)
        score += 350;
    else if (rtt_us > 50000)
        score += 280;
    else if (rtt_us > 20000)
        score += 200;
    else if (rtt_us > 5000)
        score += 100;
    
    // Jitter component (0-200 points)
    u32 jitter_us = net_mdev_us >> 3;
    if (jitter_us > 20000)
        score += 200;
    else if (jitter_us > 10000)
        score += 150;
    else if (jitter_us > 5000)
        score += 100;
    
    // Loss component (0-300 points)
    if (net_retrans_out > 5 || net_lost > 5)
        score += 300;
    else if (net_total_retrans > 20)
        score += 200;
    else if (net_total_retrans > 5)
        score += 100;
    
    // Cwnd pressure (0-150 points)
    if (net_packets_out > 0 && net_snd_cwnd > 0) {
        if (net_packets_out * 2 > net_snd_cwnd)
            score += 150;
        else if (net_packets_out * 3 > net_snd_cwnd)
            score += 75;
    }
    
    return (score > 1000) ? 1000 : score;
}

// Update burst state machine
static __always_inline void update_burst_state(u64 now, u32 phase)
{
    u64 inter_arrival = (last_packet_ts > net_prev_packet_ts) ? 
        (last_packet_ts - net_prev_packet_ts) : 1000000000ULL;
    bool is_fast = (inter_arrival < BURST_THRESHOLD_NS);
    
    if (is_fast && phase == PHASE_BURST) {
        if (burst_start_ts == 0)
            burst_start_ts = now;
        in_burst = true;
    } else if (!is_fast || (now - burst_start_ts) > BURST_SUSTAIN_NS) {
        in_burst = false;
        burst_start_ts = 0;
    }
}

// -- Adaptive time window computation based on phase --
static __always_inline u64 get_phase_based_recent_window(u32 phase, u32 net_score)
{
    u64 base = BASE_RECENT_NS;
    
    if (phase == PHASE_IDLE)
        return base / 2;  // Aggressive during idle
    else if (phase == PHASE_NORMAL)
        return base;
    else if (phase == PHASE_BURST)
        return base * 2;  // Protect more during bursts
    else  // PHASE_CONGESTED
        return base * (1 + net_score / 500);  // Scale with congestion
    
    return base;
}

static __always_inline u64 get_phase_based_working_set(u32 phase, u32 net_score)
{
    u64 base = BASE_WORKING_SET_NS;
    
    if (phase == PHASE_IDLE)
        return base / 2;
    else if (phase == PHASE_NORMAL)
        return base;
    else if (phase == PHASE_BURST)
        return base * 2;  // Extend protection during bursts
    else  // PHASE_CONGESTED
        return base * (1 + net_score / 200);  // Extend significantly
    
    return base;
}

// -- Sampling callback: compute eviction score for a single folio --
// This is called on sampled folios, not all folios, reducing CPU overhead.
static int bpf_evict_sampled_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // Compute folio age
    u64 age_ns = (now > meta->last_access_ts) ? 
        (now - meta->last_access_ts) : 0;

    // Get phase-aware windows
    u64 recent_window = get_phase_based_recent_window(current_phase, phase_score);
    u64 working_set_window = get_phase_based_working_set(current_phase, phase_score);

    // Phase-specific eviction logic
    switch (current_phase) {
    case PHASE_IDLE:
        // Idle: aggressive eviction of everything not recently used
        if (age_ns > recent_window)
            return CACHE_EXT_EVICT_NODE;
        break;

    case PHASE_NORMAL:
        // Normal: balanced protection
        if (meta->access_tier == 0 && age_ns > recent_window)
            return CACHE_EXT_EVICT_NODE;
        if (meta->access_tier <= 1 && age_ns > working_set_window)
            return CACHE_EXT_EVICT_NODE;
        break;

    case PHASE_BURST:
        // Burst: protect working set aggressively
        if (meta->access_tier == 0 && age_ns > (recent_window * 2))
            return CACHE_EXT_EVICT_NODE;
        if (meta->access_tier == 1 && age_ns > working_set_window)
            return CACHE_EXT_EVICT_NODE;
        // Hot (tier 2) protected unless very old
        if (meta->access_tier == 2 && age_ns > (working_set_window * 2))
            return CACHE_EXT_EVICT_NODE;
        break;

    case PHASE_CONGESTED:
        // Congestion: very conservative, protect everything in working set
        if (age_ns < working_set_window)
            return CACHE_EXT_CONTINUE_ITER;
        // Only evict very old pages
        if (age_ns > (working_set_window * 3))
            return CACHE_EXT_EVICT_NODE;
        // Otherwise, protect hot/warm
        if (meta->access_tier >= 1)
            return CACHE_EXT_CONTINUE_ITER;
        break;
    }

    return CACHE_EXT_CONTINUE_ITER;
}

// -- struct_ops: init ---------------------------------------------------------
static u64 main_list;

s32 BPF_STRUCT_OPS_SLEEPABLE(evolved_init, struct mem_cgroup *memcg)
{
    main_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!main_list)
        return -1;
    return 0;
}

// -- struct_ops: evict_folios -------------------------------------------------
// Uses stratified sampling with phase-aware scoring
void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    u64 now = bpf_ktime_get_ns();
    
    bump_counter(0);

    // Compute phase once at eviction time
    current_phase = classify_phase(now);
    phase_score = compute_network_score();
    update_burst_state(now, current_phase);

    // Adaptive sampling ratio based on phase
    u32 sample_ratio = 100;  // default: sample 1%
    if (current_phase == PHASE_IDLE)
        sample_ratio = 200;  // 0.5% idle = more aggressive
    else if (current_phase == PHASE_BURST)
        sample_ratio = 50;   // 2% burst = conservative
    else if (current_phase == PHASE_CONGESTED)
        sample_ratio = 30;   // 3.3% congested = very conservative

    // Use sampling instead of full iteration to reduce CPU overhead
    bpf_cache_ext_list_sample(memcg, main_list, bpf_evict_sampled_cb, 
                              sample_ratio, eviction_ctx);
}

// -- struct_ops: folio_added --------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_added, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    int ret = bpf_cache_ext_list_add_tail(main_list, folio);
    if (ret != 0) {
        bump_counter(1);
        return;
    }

    u64 key = (u64)folio;
    struct net_folio_metadata new_meta = {
        .last_access_ts = bpf_ktime_get_ns(),
        .access_count  = 1,
        .access_tier   = 0,
    };
    bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
    bump_counter(2);
}

// -- struct_ops: folio_accessed -----------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 key = (u64)folio;
    u64 now = bpf_ktime_get_ns();

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta) {
        int ret = bpf_cache_ext_list_add(main_list, folio);
        if (ret != 0) {
            bpf_cache_ext_list_move(main_list, folio, false);
        }
        struct net_folio_metadata new_meta = {
            .last_access_ts = now,
            .access_count  = 1,
            .access_tier   = 0,
        };
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    // Update metadata
    meta->last_access_ts = now;
    meta->access_count += 1;

    // Update tier
    u32 old_tier = meta->access_tier;
    if (meta->access_count >= 4)
        meta->access_tier = 2;
    else if (meta->access_count >= 2)
        meta->access_tier = 1;
    else
        meta->access_tier = 0;

    // Move to head (MRU)
    bpf_cache_ext_list_move(main_list, folio, false);
    bump_counter(4);
}

// -- struct_ops: folio_evicted ------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_evicted, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    bpf_cache_ext_list_del(folio);

    u64 key = (u64)folio;
    bpf_map_delete_elem(&folio_metadata_map, &key);
    bump_counter(5);
}

SEC("kprobe/tcp_recvmsg")
int trace_tcp_recvmsg(struct pt_regs *ctx)
{
    struct sock *sk = (struct sock *)PT_REGS_PARM1(ctx);
    if (!sk) return 0;

    u16 sport = BPF_CORE_READ(sk, __sk_common.skc_num);

    if (sport == 9100 || sport == 9001 || sport == 9002) {
        net_prev_packet_ts = last_packet_ts;
        last_packet_ts = bpf_ktime_get_ns();
        net_recv_count += 1;

        struct tcp_sock *tp = (struct tcp_sock *)sk;

        net_srtt_us        = BPF_CORE_READ(tp, srtt_us);
        net_mdev_us        = BPF_CORE_READ(tp, mdev_us);
        net_bytes_received = BPF_CORE_READ(tp, bytes_received);
        net_bytes_acked    = BPF_CORE_READ(tp, bytes_acked);
        net_segs_in        = BPF_CORE_READ(tp, segs_in);
        net_segs_out       = BPF_CORE_READ(tp, segs_out);
        net_delivered      = BPF_CORE_READ(tp, delivered);
        net_snd_cwnd       = BPF_CORE_READ(tp, snd_cwnd);
        net_snd_ssthresh   = BPF_CORE_READ(tp, snd_ssthresh);
        net_rcv_wnd        = BPF_CORE_READ(tp, rcv_wnd);
        net_packets_out    = BPF_CORE_READ(tp, packets_out);
        net_total_retrans  = BPF_CORE_READ(tp, total_retrans);
        net_retrans_out    = BPF_CORE_READ(tp, retrans_out);
        net_lost           = BPF_CORE_READ(tp, lost);
    }
    return 0;
}
