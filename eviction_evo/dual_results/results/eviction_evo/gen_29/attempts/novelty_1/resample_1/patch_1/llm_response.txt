<DESCRIPTION>
I drew inspiration from the context programs' multi-tier architecture and network-aware eviction strategies, but I'm implementing a fundamentally different approach to tier promotion and eviction scoring.

Key insights from the context programs:
1. The best programs use tier-specific adaptive windows that scale with congestion
2. Burst detection and working-set protection are critical
3. Virtual decay (for scoring) vs actual access_count (for promotion) separation works well
4. Exponential decay models provide smooth aging behavior

My novel approach:
1. **Dynamic Tier Promotion Thresholds**: Instead of fixed access_count thresholds (2 for warm, 4 for hot), I implement congestion-adaptive promotion thresholds that lower during extreme congestion (when cache misses are most expensive) and raise during low congestion (to maintain aggressive eviction). This directly addresses the recommendation to "implement congestion-aware tier promotion thresholds."

2. **Hybrid Scoring Model**: I combine exponential decay (from the best-performing program #4) with tier-specific adaptive windows (from programs #2-3) and add a novel "congestion pressure multiplier" that amplifies protection during network stress.

3. **Adaptive Eviction Thresholds**: Rather than fixed score thresholds per tier, I compute dynamic eviction thresholds that scale with both congestion AND burst state, providing finer-grained control.

4. **Burst-Aware Promotion**: During bursts, I temporarily lower promotion thresholds to rapidly identify the working set, then restore normal thresholds when burst ends.

This design maintains the proven multi-tier architecture but fundamentally changes how folios move between tiers and how eviction decisions are made, creating a more responsive system that adapts to both network conditions and access patterns.
</DESCRIPTION>

// -- Tunable constants --------------------------------------------------------
// Tier-specific exponential decay half-lives
#define DECAY_HALF_LIFE_COLD_NS  80000000ULL    // 80 ms (cold decays fast)
#define DECAY_HALF_LIFE_WARM_NS  350000000ULL   // 350 ms (warm moderate)
#define DECAY_HALF_LIFE_HOT_NS   750000000ULL   // 750 ms (hot decays slow)

// Network idle timeout
#define NET_IDLE_TIMEOUT_NS      5000000000ULL  // 5 s
// Burst detection threshold
#define BURST_THRESHOLD_NS       5000000ULL     // 5 ms

// Base tier promotion thresholds (will be dynamically adjusted)
#define BASE_WARM_THRESHOLD      2
#define BASE_HOT_THRESHOLD       4

// Eviction score thresholds (base values, scaled by congestion and burst)
#define BASE_EVICT_THRESHOLD_COLD   120
#define BASE_EVICT_THRESHOLD_WARM   180
#define BASE_EVICT_THRESHOLD_HOT    80

// Congestion scaling factors
#define CONGESTION_DECAY_SCALE      400ULL      // dampen decay during congestion
#define CONGESTION_THRESHOLD_SCALE  600ULL      // raise eviction thresholds during congestion
#define BURST_DECAY_MULTIPLIER      8ULL        // freeze decay during bursts

// -- Helper: Compute network congestion score (0-1000 scale) --
static __always_inline u32 compute_congestion_score(void) {
    u32 score = 0;

    u32 rtt_us = net_srtt_us >> 3;
    if (rtt_us > 100000)
        score += 400;
    else if (rtt_us > 50000)
        score += 300;
    else if (rtt_us > 20000)
        score += 200;
    else if (rtt_us > 5000)
        score += 100;

    u32 mdev_us = net_mdev_us >> 3;
    if (mdev_us > 20000)
        score += 250;
    else if (mdev_us > 10000)
        score += 200;
    else if (mdev_us > 5000)
        score += 150;
    else if (mdev_us > 1000)
        score += 100;

    if (net_retrans_out > 5 || net_lost > 0)
        score += 200;
    else if (net_total_retrans > 10)
        score += 150;
    else if (net_total_retrans > 0)
        score += 50;

    if (net_packets_out > 0 && net_snd_cwnd > 0) {
        if (net_packets_out * 2 > net_snd_cwnd)
            score += 150;
        else if (net_packets_out * 3 > net_snd_cwnd)
            score += 100;
    }

    return (score > 1000) ? 1000 : score;
}

// -- Helper: Detect burst mode --
static __always_inline bool is_burst(u64 now) {
    if (net_prev_packet_ts == 0)
        return false;
    u64 inter_arrival_ns = (now > net_prev_packet_ts) ?
        (now - net_prev_packet_ts) : 0;
    return (inter_arrival_ns < BURST_THRESHOLD_NS);
}

// -- Helper: Get dynamic tier promotion thresholds based on congestion and burst --
// NOVEL: Adapts promotion thresholds to network conditions
static __always_inline void get_dynamic_promotion_thresholds(u32 congestion,
                                                             bool in_burst,
                                                             u32 *warm_threshold,
                                                             u32 *hot_threshold) {
    // During extreme congestion (>700), lower thresholds to rapidly identify working set
    if (congestion > 700) {
        *warm_threshold = 1;   // promote to warm at 1 access
        *hot_threshold = 2;    // promote to hot at 2 accesses
    }
    // During high congestion (500-700), moderately lower thresholds
    else if (congestion > 500) {
        *warm_threshold = 1;
        *hot_threshold = 3;
    }
    // During medium congestion (300-500), balanced approach
    else if (congestion > 300) {
        *warm_threshold = 2;
        *hot_threshold = 4;
    }
    // During low congestion (<300), raise thresholds for aggressive eviction
    else {
        *warm_threshold = 3;
        *hot_threshold = 5;
    }

    // Burst override: during bursts, temporarily lower thresholds to capture working set
    if (in_burst) {
        if (*warm_threshold > 1)
            *warm_threshold -= 1;
        if (*hot_threshold > 2)
            *hot_threshold -= 1;
    }
}

// -- Helper: Compute adaptive decay half-life based on congestion and burst --
static __always_inline u64 get_adaptive_decay_half_life(u32 tier, u32 congestion, bool in_burst) {
    u64 base_half_life;
    
    if (tier == 2)
        base_half_life = DECAY_HALF_LIFE_HOT_NS;
    else if (tier == 1)
        base_half_life = DECAY_HALF_LIFE_WARM_NS;
    else
        base_half_life = DECAY_HALF_LIFE_COLD_NS;

    // Burst protection: multiply half-life to freeze decay
    if (in_burst) {
        base_half_life = base_half_life * BURST_DECAY_MULTIPLIER;
        if (base_half_life > 6000000000ULL)  // cap at 6s
            base_half_life = 6000000000ULL;
    }

    // Congestion damping: scale from base to base * 2.5 based on congestion
    u64 congestion_factor = (congestion * 1500) / CONGESTION_DECAY_SCALE;
    if (congestion_factor > 1500)
        congestion_factor = 1500;
    
    base_half_life = base_half_life + (base_half_life * congestion_factor) / 1000;

    return base_half_life;
}

// -- Helper: Compute exponential decay score --
static __always_inline u32 compute_decay_score(u64 age_ns, u64 half_life_ns) {
    if (age_ns == 0)
        return 1000;
    
    if (age_ns > half_life_ns * 10)
        return 0;
    
    u64 decay_factor = (age_ns * 256) / half_life_ns;
    
    u32 score = 1000;
    
    if (decay_factor >= 256) {
        score = score >> 1;
        decay_factor -= 256;
    }
    if (decay_factor >= 128) {
        score = score >> 1;
        decay_factor -= 128;
    }
    if (decay_factor >= 64) {
        score = score >> 1;
        decay_factor -= 64;
    }
    if (decay_factor >= 32) {
        score = score >> 1;
        decay_factor -= 32;
    }
    if (decay_factor >= 16) {
        score = score >> 1;
        decay_factor -= 16;
    }
    if (decay_factor >= 8) {
        score = score >> 1;
        decay_factor -= 8;
    }
    if (decay_factor >= 4) {
        score = score >> 1;
        decay_factor -= 4;
    }
    if (decay_factor >= 2) {
        score = score >> 1;
        decay_factor -= 2;
    }
    if (decay_factor >= 1) {
        score = score - (score >> 1);
    }

    return score;
}

// -- Helper: Compute frequency bonus --
static __always_inline u32 compute_frequency_bonus(u32 access_count) {
    if (access_count >= 6)
        return 400;
    else if (access_count >= 5)
        return 350;
    else if (access_count >= 4)
        return 300;
    else if (access_count >= 3)
        return 200;
    else if (access_count >= 2)
        return 100;
    else
        return 0;
}

// -- Helper: Get dynamic eviction threshold based on congestion, burst, and tier --
// NOVEL: Combines multiple factors for adaptive threshold
static __always_inline u32 get_dynamic_evict_threshold(u32 tier, u32 congestion, bool in_burst) {
    u32 base_threshold;
    
    if (tier == 2)
        base_threshold = BASE_EVICT_THRESHOLD_HOT;
    else if (tier == 1)
        base_threshold = BASE_EVICT_THRESHOLD_WARM;
    else
        base_threshold = BASE_EVICT_THRESHOLD_COLD;

    // Congestion adjustment: raise threshold (keep more) during congestion
    u32 congestion_adjustment = (congestion * base_threshold) / CONGESTION_THRESHOLD_SCALE;
    
    // Burst adjustment: raise threshold (keep more) during bursts
    u32 burst_adjustment = 0;
    if (in_burst) {
        burst_adjustment = base_threshold / 3;  // add 33% during bursts
    }

    u32 final_threshold = base_threshold + congestion_adjustment + burst_adjustment;
    
    // Cap to reasonable range
    if (final_threshold > 800)
        final_threshold = 800;
    
    return final_threshold;
}

// -- Helper: Compute congestion pressure multiplier --
// NOVEL: Amplifies protection during network stress
static __always_inline u32 compute_congestion_pressure_multiplier(u32 congestion, bool session_active) {
    if (!session_active)
        return 0;
    
    // Scale from 0 to 150 points based on congestion severity
    if (congestion > 700)
        return 150;
    else if (congestion > 500)
        return 100;
    else if (congestion > 300)
        return 50;
    else
        return 0;
}

// -- Eviction callback for cold_list --
static int bpf_evict_cold_cb(int idx, struct cache_ext_list_node *node) {
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age_ns = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u32 congestion = compute_congestion_score();
    bool in_burst = is_burst(now);
    bool session_active = (now - last_packet_ts) < NET_IDLE_TIMEOUT_NS;

    u64 half_life = get_adaptive_decay_half_life(0, congestion, in_burst);
    u32 decay_score = compute_decay_score(age_ns, half_life);
    u32 freq_bonus = compute_frequency_bonus(meta->access_count);
    u32 congestion_bonus = compute_congestion_pressure_multiplier(congestion, session_active);
    
    u32 total_score = decay_score + freq_bonus + congestion_bonus;

    u32 threshold = get_dynamic_evict_threshold(0, congestion, in_burst);

    if (total_score < threshold)
        return CACHE_EXT_EVICT_NODE;

    return CACHE_EXT_CONTINUE_ITER;
}

{
    hot_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!hot_list)
        return -1;

    if (!warm_list)
        return -1;

    cold_list = bpf_cache_ext_ds_registry_new_list(memcg);
    if (!cold_list)

    return 0;
}

// -- struct_ops: evict_folios -------------------------------------------------
void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    bump_counter(0);

    bpf_cache_ext_list_iterate(memcg, cold_list, bpf_evict_cold_cb, eviction_ctx);
    bpf_cache_ext_list_iterate(memcg, warm_list, bpf_evict_warm_cb, eviction_ctx);
    bpf_cache_ext_list_iterate(memcg, hot_list, bpf_evict_hot_cb, eviction_ctx);
}

// -- struct_ops: folio_added --------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_added, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 key = (u64)folio;

    int ret = bpf_cache_ext_list_add_tail(cold_list, folio);
    if (ret != 0) {
        bump_counter(1);
        return;
    }

    struct net_folio_metadata new_meta = {
        .last_access_ts = bpf_ktime_get_ns(),
        .access_count  = 1,
        .access_tier   = 0,
    };
    bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
    bump_counter(2);
}

// -- struct_ops: folio_accessed -----------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_accessed, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    u64 key = (u64)folio;
    u64 now = bpf_ktime_get_ns();

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta) {
        bpf_cache_ext_list_add_tail(cold_list, folio);
        struct net_folio_metadata new_meta = {
            .last_access_ts = now,
            .access_count  = 1,
            .access_tier   = 0,
        };
        bpf_map_update_elem(&folio_metadata_map, &key, &new_meta, BPF_ANY);
        bump_counter(3);
        return;
    }

    meta->last_access_ts = now;
    meta->access_count += 1;

    // Get dynamic promotion thresholds based on current network conditions
    u32 congestion = compute_congestion_score();
    bool in_burst = is_burst(now);
    u32 warm_threshold, hot_threshold;
    get_dynamic_promotion_thresholds(congestion, in_burst, &warm_threshold, &hot_threshold);

    // Determine new tier using dynamic thresholds
    u32 old_tier = meta->access_tier;
    if (meta->access_count >= hot_threshold)
        meta->access_tier = 2;
    else if (meta->access_count >= warm_threshold)
        meta->access_tier = 1;
    else
        meta->access_tier = 0;

    if (meta->access_tier > old_tier) {
        bpf_cache_ext_list_del(folio);

        if (meta->access_tier == 2) {
            bpf_cache_ext_list_add(hot_list, folio);
        } else if (meta->access_tier == 1) {
            bpf_cache_ext_list_add(warm_list, folio);
        }
    } else {
        if (meta->access_tier == 2)
            bpf_cache_ext_list_move(hot_list, folio, false);
        else if (meta->access_tier == 1)
            bpf_cache_ext_list_move(warm_list, folio, false);
        else
            bpf_cache_ext_list_move(cold_list, folio, false);
    }

    bump_counter(4);
}

// -- struct_ops: folio_evicted ------------------------------------------------
void BPF_STRUCT_OPS(evolved_folio_evicted, struct folio *folio)
{
    if (!is_folio_relevant(folio))
        return;

    bpf_cache_ext_list_del(folio);

    u64 key = (u64)folio;
    bpf_map_delete_elem(&folio_metadata_map, &key);
    bump_counter(5);
}

SEC("kprobe/tcp_recvmsg")
int trace_tcp_recvmsg(struct pt_regs *ctx)
{
    struct sock *sk = (struct sock *)PT_REGS_PARM1(ctx);
    if (!sk) return 0;

    u16 sport = BPF_CORE_READ(sk, __sk_common.skc_num);

    if (sport == 9100 || sport == 9001 || sport == 9002) {
        net_prev_packet_ts = last_packet_ts;
        last_packet_ts = bpf_ktime_get_ns();
        net_recv_count += 1;

        struct tcp_sock *tp = (struct tcp_sock *)sk;

        net_srtt_us        = BPF_CORE_READ(tp, srtt_us);
        net_mdev_us        = BPF_CORE_READ(tp, mdev_us);

        net_bytes_received = BPF_CORE_READ(tp, bytes_received);
        net_bytes_acked    = BPF_CORE_READ(tp, bytes_acked);
        net_segs_in        = BPF_CORE_READ(tp, segs_in);
        net_segs_out       = BPF_CORE_READ(tp, segs_out);
        net_delivered      = BPF_CORE_READ(tp, delivered);

        net_snd_cwnd       = BPF_CORE_READ(tp, snd_cwnd);
        net_snd_ssthresh   = BPF_CORE_READ(tp, snd_ssthresh);
        net_rcv_wnd        = BPF_CORE_READ(tp, rcv_wnd);
        net_packets_out    = BPF_CORE_READ(tp, packets_out);

        net_total_retrans  = BPF_CORE_READ(tp, total_retrans);
        net_retrans_out    = BPF_CORE_READ(tp, retrans_out);
        net_lost           = BPF_CORE_READ(tp, lost);
    }
    return 0;
}
