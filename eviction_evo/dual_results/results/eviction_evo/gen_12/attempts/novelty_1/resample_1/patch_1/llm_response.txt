// Higher congestion → slower decay (wider window)
static __always_inline u64 get_adaptive_decay_window(u32 congestion) {
    // Base 500ms, scale to 1.5s at max congestion
    u64 base = BASE_DECAY_WINDOW_NS;
    u64 window = base + (1000000000ULL * congestion) / 1000;  // add 0-1s
    return (window > 1500000000ULL) ? 1500000000ULL : window;

// -- Helper: Apply access count decay based on folio age and tier --
// Decays access_count if folio hasn't been accessed recently
// Returns the decayed access_count
static __always_inline u32 apply_access_count_decay(struct net_folio_metadata *meta,
                                                     u64 now, u32 congestion) {
    if (!meta || meta->access_count <= 1)
        return meta ? meta->access_count : 1;

    u64 age_ns = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u64 decay_window = get_adaptive_decay_window(congestion);

    u64 tier_sensitivity = DECAY_SENSITIVITY_WARM;  // default

        tier_sensitivity = DECAY_SENSITIVITY_HOT;

    // Scale decay window by tier sensitivity
    u64 tier_decay_window = (decay_window * tier_sensitivity) / 1500ULL;

    // If age exceeds tier-specific decay window, decay access_count by 1

        if (decayed_count < 1)

        return decayed_count;
    }

    return meta->access_count;
}
