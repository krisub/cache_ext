<DESCRIPTION>
The current program's `access_count` only increases monotonically, causing folios that were frequently accessed in the past but are now cold to remain protected indefinitely. This is problematic during workload phase transitions (e.g., when the dual_small_vs_large trace switches from one client's working set to another's).

The recommendation suggests implementing frequency-decay scoring to penalize stale high-access-count folios. I'll implement this by:

1. Adding a decay mechanism in the eviction callbacks that reduces the effective access_count based on age
2. The decay will be tier-specific: cold tier decays faster, hot tier decays slower
3. During scoring, we'll compute an "effective access count" that decays exponentially with age
4. This complements the existing burst-detection logic by enabling responsive adaptation to sustained workload changes

The key insight is that `access_count` should represent recent frequency, not lifetime frequency. By decaying it based on time since last access, we allow the eviction policy to adapt when the working set changes, while still protecting truly hot pages that are accessed frequently.

Implementation approach:
- In each eviction callback, compute an effective_access_count = access_count * decay_factor(age)
- Use this effective count in the frequency bonus calculation instead of raw access_count
- The decay factor will use the same half-life approach as the recency decay, but applied to frequency

This should improve p99 latencies (currently 234ms for dual_small_vs_large) by preventing stale high-frequency folios from blocking eviction of newly-active working sets during workload transitions.

// -- Helper: Compute effective access count with frequency decay --
// Decays access_count based on age to penalize stale high-frequency folios
static __always_inline u32 compute_effective_access_count(u32 access_count, u64 age_ns, u32 tier) {
    if (access_count <= 1)
        return access_count;

    // Frequency decay half-life (time for access_count to halve)
    // Shorter than recency decay to more aggressively forget old frequency
    u64 freq_decay_half_life;
    if (tier == 2)  // hot
        freq_decay_half_life = 600000000ULL;  // 600ms
    else if (tier == 1)  // warm
        freq_decay_half_life = 300000000ULL;  // 300ms
    else  // cold
        freq_decay_half_life = 150000000ULL;  // 150ms

    // Compute decay factor (0-256 scale, where 256 = no decay)
    u32 decay_factor = 256;
    if (age_ns > freq_decay_half_life) {
        u64 num_half_lives = age_ns / freq_decay_half_life;
        // Each half-life halves the count
        if (num_half_lives >= 8)
            decay_factor = 1;  // Nearly zero after 8 half-lives
        else
            decay_factor = 256 >> num_half_lives;
    } else {
        // Linear interpolation for fractional half-life
        u64 fraction = (age_ns * 256) / freq_decay_half_life;
        decay_factor = 256 - (fraction >> 1);  // Approximate exponential
    }

    // Apply decay to access_count
    u32 effective_count = (access_count * decay_factor) / 256;

    // Ensure minimum of 1 if original was > 0
    return (effective_count == 0 && access_count > 0) ? 1 : effective_count;
}

// -- Helper: Compute frequency bonus (compounds with effective access count) --
static __always_inline u32 compute_frequency_bonus(u32 effective_access_count) {
    if (effective_access_count >= 6)
        return 400;

        return 350;

        return 300;
    else if (effective_access_count >= 3)
        return 200;
    else if (effective_access_count >= 2)
        return 100;

        return 0;
}

// -- Eviction callback for cold_list --
static int bpf_evict_cold_cb(int idx, struct cache_ext_list_node *node) {
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    u64 age_ns = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u32 congestion = compute_congestion_score();
    bool in_burst = is_burst(now);

    u64 half_life = get_adaptive_decay_half_life(0, congestion, in_burst);
    u32 decay_score = compute_decay_score(age_ns, half_life);
    u32 freq_bonus = compute_frequency_bonus(meta->access_count);
    u32 total_score = decay_score + freq_bonus;

    u32 threshold = get_evict_threshold(0, congestion);

    if (total_score < threshold)
        return CACHE_EXT_EVICT_NODE;

    return CACHE_EXT_CONTINUE_ITER;
}
