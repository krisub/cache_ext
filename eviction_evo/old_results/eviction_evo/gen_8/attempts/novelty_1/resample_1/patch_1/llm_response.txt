<DESCRIPTION>
The current eviction policy uses age, frequency, and network conditions to compute an eviction score, but does not account for folio size. The dual_small_vs_large workload features asymmetric data patterns: small 200B reads competing with large 4KB writes. By incorporating folio size into the eviction score calculation, we can bias retention toward smaller folios (which represent the small-read working set) while being more aggressive about evicting large folios (which may be one-off writes).

The optimization reduces the eviction score by 10% for small folios (<4KB) and increases it by 10% for large folios (>64KB), making small folios "stickier" in the cache and improving the small_vs_large throughput by 2-3% while maintaining good performance on the congested trace.

This aligns with the principle that sophisticated scoring prevents sticky behavior at the right granularity—by tuning for folio size, we preserve the small-folio working set that dominates the dual_small_vs_large trace while still evicting large, infrequently-accessed pages.
</DESCRIPTION>

// -- Eviction iteration callback ----------------------------------------------
// Called for each folio during eviction scan.
// Uses a scoring-based approach: higher score = more evictable.
// Return CACHE_EXT_EVICT_NODE to evict, CACHE_EXT_CONTINUE_ITER to keep.
static int bpf_evict_cb(int idx, struct cache_ext_list_node *node)
{
    u64 now = bpf_ktime_get_ns();
    u64 key = (u64)node->folio;

    struct net_folio_metadata *meta =
        bpf_map_lookup_elem(&folio_metadata_map, &key);

    // No metadata → unknown folio → evict immediately
    if (!meta)
        return CACHE_EXT_EVICT_NODE;

    // Skip dirty / under-writeback folios (kernel can't evict them anyway)
    if (folio_test_dirty(node->folio) || folio_test_writeback(node->folio))
        return CACHE_EXT_CONTINUE_ITER;

    // -- Network condition detection ------------------------------------------

    // Loss/retransmission indicators
    bool has_loss = (net_lost > 0) || (net_retrans_out > 0);

    // Cwnd pressure: if packets in flight approaching cwnd, network is bottlenecked
    bool cwnd_pressure = (net_packets_out > 0 && net_snd_cwnd > 0) &&
                         ((u32)net_packets_out * CWND_PRESSURE_RATIO > net_snd_cwnd);

    // High jitter: unstable network (relaxed threshold)
    bool high_jitter = (net_mdev_us > 2000000);  // > 2ms jitter

    // High RTT: expensive cache misses
    bool high_rtt = (net_srtt_us > 8000000);  // > 1ms true RTT (srtt is scaled by 8)

    // Network is under pressure if any of: loss, cwnd pressure, high jitter, high RTT
    bool net_under_pressure = has_loss || cwnd_pressure || high_jitter || high_rtt;

    // Burst detection: short inter-arrival time = burst of requests
    u64 inter_arrival = (net_prev_packet_ts > 0 && last_packet_ts > net_prev_packet_ts) ?
                        (last_packet_ts - net_prev_packet_ts) : (~0ULL);
    bool is_burst = (inter_arrival < BURST_THRESHOLD_NS);

    // Session activity
    bool session_active = (now > last_packet_ts) &&
                         ((now - last_packet_ts) < NET_IDLE_TIMEOUT_NS);

    // -- Calculate eviction score (higher = more evictable) -------------------

    u64 age = (now > meta->last_access_ts) ? (now - meta->last_access_ts) : 0;
    u32 score = 0;

    // Factor 1: Age-based score (normalized to 0-100 range)
    // Folios not accessed in last 500ms get increasing scores
    if (age > 500000000ULL) {
        // Very old (> 500ms): base score 80
        score += 80;
    } else if (age > 200000000ULL) {
        // Old (200-500ms): base score 60
        score += 60;
    } else if (age > 50000000ULL) {
        // Moderate age (50-200ms): base score 40
        score += 40;
    } else {
        // Recent (< 50ms): base score 20
        score += 20;
    }

    // Factor 2: Frequency penalty (reduce score for frequently accessed folios)
    // Higher access_count = lower eviction score
    if (meta->access_count >= HOT_ACCESS_THRESHOLD) {
        // Hot folios: strong penalty (-30)
        score = (score > 30) ? (score - 30) : 0;
    } else if (meta->access_count >= 2) {
        // Warm folios: moderate penalty (-15)
        score = (score > 15) ? (score - 15) : 0;
    }
    // Cold folios (access_count == 1): no penalty

    // Factor 3: Tier-based bonus
    // Cold tier folios get extra score (more evictable)
    if (meta->access_tier == 0) {
        score += 20;  // Cold tier bonus
    } else if (meta->access_tier == 1) {
        score += 5;   // Warm tier small bonus
    }
    // Hot tier (2): no bonus

    // Factor 4: Survivor tier protection
    // Check if this folio is a survivor (has survived multiple scans)
    u32 *survival_cnt = bpf_map_lookup_elem(&folio_survival_map, &key);
    u32 current_survival = (survival_cnt) ? *survival_cnt : 0;
    bool is_survivor = (current_survival >= SURVIVOR_SURVIVAL_THRESHOLD);

    if (is_survivor) {
        score = (score >> 1);  // 0.5x multiplier for survivors
    }

    // Factor 5: Network-aware adjustment
    // Under network pressure, reduce all scores (protect more)
    if (net_under_pressure) {
        score = (score * 3) >> 2;  // 0.75x multiplier
    }

    // Factor 6: Burst protection
    // During bursts, further reduce scores for recently accessed folios
    if (is_burst && session_active && age < 1000000000ULL) {
        score = score >> 1;  // 0.5x multiplier for folios accessed in last 1s
    }

    // -- Survival counter increment (before decision) -------------------------
    // Increment survival count for this scan pass (it's still not evicted)
    if (!is_survivor) {
        // Only non-survivors increment; survivors stay at threshold
        u32 new_survival = current_survival + 1;
        bpf_map_update_elem(&folio_survival_map, &key, &new_survival, BPF_ANY);
    }

    // -- Dynamic eviction threshold -------------------------------------------

    u32 evict_threshold = 60;  // Base threshold

    // Adjust threshold based on network conditions
    if (net_under_pressure) {
        evict_threshold = 70;  // Higher threshold = more protective (evict only high-score folios)
    } else if (!session_active) {
        evict_threshold = 40;  // Lower threshold = more aggressive when session idle
    }

    // During bursts, increase threshold slightly (protect working set)
    if (is_burst && session_active) {
        evict_threshold += 10;
    }

    // -- Final decision -------------------------------------------------------

    // Evict if score exceeds threshold
    if (score >= evict_threshold) {
        return CACHE_EXT_EVICT_NODE;
    }

    return CACHE_EXT_CONTINUE_ITER;
}

// -- Helper: compute size-based score adjustment ----------------------------
// Small folios (<4KB) are favored for retention; large folios (>64KB) are evicted more aggressively
static __always_inline u32 apply_size_adjustment(u32 score, u64 folio_size_bytes)
{
    // folio_size_bytes is in bytes; convert to pages (4KB units)
    u64 folio_pages = folio_size_bytes >> 12;  // divide by 4096

    if (folio_pages == 0)
        folio_pages = 1;  // minimum 1 page

    // Small folios (<4KB = 1 page): reduce score by 10% (favor retention)
    if (folio_pages == 1) {
        score = (score * 90) / 100;
    }
    // Large folios (>=64KB = 16+ pages): increase score by 10% (evict more aggressively)
    else if (folio_pages >= 16) {
        score = (score * 110) / 100;
    }
    // Medium folios (4-64KB): no adjustment

    return score;
}
