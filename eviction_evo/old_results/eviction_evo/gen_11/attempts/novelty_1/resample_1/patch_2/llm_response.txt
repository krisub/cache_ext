<DESCRIPTION>
The current program lacks sustained burst detection, which is a missed opportunity for network-aware adaptation. The best performing program (score 2162.27) used burst detection based on inter-arrival time, but this only captures instantaneous spikes.

The recommendation suggests adding a complementary **packet count velocity** metric within the EVOLVE-BLOCK: tracking how many tcp_recvmsg calls occur within a 100ms sliding window. This distinguishes:
- **Transient spikes**: high frequency packets in a short window (inter-arrival < 10ms)
- **Sustained bursts**: consistently high packet rate over 100ms (net_recv_count increasing by 15+ packets)

During sustained bursts, protecting the working set becomes critical for P99 latency. By adding packet velocity detection and applying stronger protection to recently-accessed folios (age < 500ms) during sustained bursts, we can better protect the active working set during congested scenarios while remaining aggressive during idle periods.

This should improve the dual_congested trace (where 2ms network delays cause sustained bursts) while maintaining or improving the dual_small_vs_large trace.
</DESCRIPTION>

// -- Helper: compute adaptive time window based on network conditions --------
static __always_inline u64 get_adaptive_window(u64 base_ns, bool under_pressure)
{
    u64 adapted = base_ns;

    // Scale by RTT: higher RTT = longer windows (each miss is more expensive)
    // Clamp net_srtt_us to avoid overflow; assume max ~10ms RTT for scaling
    u32 rtt_clamped = (net_srtt_us > 10000000) ? 10000000 : net_srtt_us;
    adapted = adapted + (u64)rtt_clamped / 20;  // Reduce RTT scaling impact

    // If network is under pressure (loss, retrans, high jitter), increase window modestly
    if (under_pressure) {
        adapted = (adapted * 5) >> 2;  // 1.25x multiplier (was 1.5x)
    }

    return adapted;
}

// -- Helper: detect sustained burst using packet velocity -----
static __always_inline bool is_sustained_burst(void)
{
    u64 now = bpf_ktime_get_ns();
    
    // Initialize burst window on first call
    if (burst_window_start_ts == 0) {
        burst_window_start_ts = now;
        burst_window_recv_count = net_recv_count;
        return false;
    }
    
    // Check if we're still within the 100ms window
    u64 window_age = (now > burst_window_start_ts) ? 
                     (now - burst_window_start_ts) : 0;
    
    if (window_age > BURST_VELOCITY_WINDOW_NS) {
        // Window expired; reset and check if we had velocity
        u64 packets_in_window = (net_recv_count >= burst_window_recv_count) ?
                                (net_recv_count - burst_window_recv_count) : 0;
        
        // Start new window
        burst_window_start_ts = now;
        burst_window_recv_count = net_recv_count;
        
        // Sustained burst: more than BURST_VELOCITY_THRESHOLD packets in last window
        return (packets_in_window > BURST_VELOCITY_THRESHOLD);
    }
    
    // Still within window; report if we're already above threshold
    u64 packets_so_far = (net_recv_count >= burst_window_recv_count) ?
                         (net_recv_count - burst_window_recv_count) : 0;
    return (packets_so_far > BURST_VELOCITY_THRESHOLD);
}
