sustained_burst_detection
</NAME>

<DESCRIPTION>
The current burst detection relies solely on inter-arrival time (< 10ms) to identify request bursts. However, this can miss sustained high-frequency workloads where requests don't have consistent sub-10ms spacing but maintain steady high packet rates.

This change adds a secondary burst signal based on packet count velocity: track the previous net_recv_count value and check if more than 15 packets arrive within a 100ms window. During sustained bursts, we apply stronger protection to folios accessed within the last 500ms by reducing their eviction scores further (0.4x multiplier instead of current implicit protection).

This complements the existing inter-arrival based burst detection and helps distinguish between transient spikes and sustained load, improving throughput during congested scenarios while maintaining good performance on asymmetric workloads.
</DESCRIPTION>

// -- struct_ops: evict_folios -------------------------------------------------
void BPF_STRUCT_OPS(evolved_evict_folios,
                    struct cache_ext_eviction_ctx *eviction_ctx,
                    struct mem_cgroup *memcg)
{
    bump_counter(0);
    
    // Update burst velocity tracking state before eviction scan
    u64 now = bpf_ktime_get_ns();
    u32 burst_state_key = 0;
    struct burst_velocity_state *burst_state = bpf_map_lookup_elem(&burst_velocity_map, &burst_state_key);
    
    if (!burst_state) {
        // First initialization
        struct burst_velocity_state init_state = {
            .prev_net_recv_count = net_recv_count,
            .last_check_ts = now,
        };
        bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &init_state, BPF_ANY);
    } else if (now > burst_state->last_check_ts) {
        u64 elapsed = now - burst_state->last_check_ts;
        if (elapsed > BURST_VELOCITY_WINDOW_NS) {
            // Window expired: update velocity baseline
            struct burst_velocity_state updated_state = {
                .prev_net_recv_count = net_recv_count,
                .last_check_ts = now,
            };
            bpf_map_update_elem(&burst_velocity_map, &burst_state_key, &updated_state, BPF_ANY);
        }
    }
    
    bpf_cache_ext_list_iterate(memcg, main_list, bpf_evict_cb, eviction_ctx);
}
